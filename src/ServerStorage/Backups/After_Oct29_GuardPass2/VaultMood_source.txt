--[[
	VaultMood: each vault level's own light. The vault is open to the sky, so on its own every level shared the same
	sunny early afternoon (a Haunted Crypt in bright daylight). As you walk from level to level this eases the time of
	day, the ambient light, the haze and a colour grade to that level's mood, and back to the lobby's look when you
	leave the vault. Only this screen's Lighting changes; the server's settings (World) are the lobby look and are
	read once at the start, so nothing is lost.

	MOOD[level] (levels without one keep the lobby look):
	  Clock    time of day (0 = midnight: the night sky and stars)    Bright   sun brightness
	  Amb/Out  Ambient / OutdoorAmbient                               Tint, Sat, Con, Exp   colour grade (Exp: a small brightness lift so night levels stay readable)
	  Haze     Atmosphere density (0 = clear, ~0.5 = thick fog) in AtmoColor
]]

local map = workspace:WaitForChild("StealAGameMap")
local starts
while not starts do
	local raw = map:GetAttribute("LevelStarts")
	if raw then
		starts = HttpService:JSONDecode(raw)
	else
		map:GetAttributeChangedSignal("LevelStarts"):Wait()
	end
end
local halfW = (map:GetAttribute("HallWidth") or 120) / 2 + 10

local grade = Instance.new("ColorCorrectionEffect")
grade.Name = "VaultMood"
grade.Parent = Lighting
local atmosphere = Lighting:FindFirstChildOfClass("Atmosphere")

-- the lobby look, as the server set it
local LOBBY = {Clock = Lighting.ClockTime, Bright = Lighting.Brightness, Amb = Lighting.Ambient, Out = Lighting.OutdoorAmbient,
	Tint = Color3.new(1, 1, 1), Sat = 0, Con = 0, Exp = 0,
	Haze = atmosphere and atmosphere.Density or 0, AtmoColor = atmosphere and atmosphere.Color or Color3.new(1, 1, 1)}

local function levelAt(pos)
	if math.abs(pos.X) > halfW or pos.Z < starts[1] or pos.Z >= starts[#starts] then
		return nil
	end
	for i = 1, #starts - 1 do
		if pos.Z < starts[i + 1] then
			return i
		end
	end
end

local current, tweens = false, {}
local function apply(mood)
	for _, tw in ipairs(tweens) do
		tw:Cancel()
	end
	local info = TweenInfo.new(1.5, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
	-- ClockTime wraps at 24: go the short way round (21 -> 0 runs forward through midnight)
	local clock = mood.Clock
	local now = Lighting.ClockTime
	if clock - now > 12 then
		clock -= 24
	elseif now - clock > 12 then
		clock += 24
	end
	tweens = {
		TweenService:Create(Lighting, info, {ClockTime = clock, Brightness = mood.Bright, Ambient = mood.Amb, OutdoorAmbient = mood.Out}),
		TweenService:Create(grade, info, {TintColor = mood.Tint, Saturation = mood.Sat, Contrast = mood.Con, Brightness = mood.Exp or 0}),
	}
	if atmosphere then
		table.insert(tweens, TweenService:Create(atmosphere, info, {Density = mood.Haze, Color = mood.AtmoColor}))
	end
	for _, tw in ipairs(tweens) do
		tw:Play()
	end
end

while true do
	local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
	if root then
		local level = levelAt(root.Position)
		local mood = level and MOOD[level] or LOBBY
		if mood ~= current then
			current = mood
			apply(mood)
		end
	end
	task.wait(0.25)
end
