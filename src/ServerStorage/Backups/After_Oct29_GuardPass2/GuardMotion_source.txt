--[[
	GuardMotion: draws every shopkeeper on this screen and plays its running sounds. What each guard does is in
	ReplicatedStorage.GuardKit (MOTION, AUDIO, FX, and the solver that keeps feet on the floor); this script feeds it.

	The server moves the giants and names what they're doing through attributes: Motion (the guard's key), Chasing,
	Returning, Act + ActAt (a one-off reaction: "taunt", "grab", "laugh", "whiff"). Every frame, for each guard near the
	camera:
	  idle      always running underneath; the head turns toward the nearest player who comes close
	  gait      blended in by real ground speed with hysteresis (on above MOVE_ON body lengths a second, off below
	            MOVE_OFF), so a guard standing at its post never runs on the spot. The stride phase advances by the
	            distance really covered, so feet never slide. Walking home is a calmer version of the chase (s.Run).
	  lean      into turns and with changes of speed
	  act       the grab, laugh, whiff or taunt takes over the whole body while it lasts (crossfading from one to the
	            next) and hands back smoothly when it ends
	  gear      wings (LeftWing/RightWing joints) follow the pose's Wing/Fold; a guard with a Carry (Kage's katana,
	            Itza's staff) moves that ONE item between its storage slot and its hand: the item's joint is blended
	            between rest (the slot) and the grip, which follows the arm's own pose every frame; Swish joints
	            (a mount's tail) sway
	  effects   GuardKit.FX: jets, water, wisps... scaled by how much the guard is moving or flying; the shark's fin
	            and wake ride the water while its body is under the deck, and the mole's mound of earth the floor
	Footfalls land where the stride says a foot comes down; each guard has its own sounds (GuardKit.AUDIO). Nothing
	plays while a guard stands, laughs or grabs; the flying and floating ones hum only while they move.
]]

local function actPose(m, name, age, p)
	if name == "grab" then
		m.Grab(p, age)
	elseif name == "laugh" then
		m.Laugh(p, age)
	elseif name == "whiff" then
		K.whiff(p, age)
	elseif name == "taunt" then
		(m.Taunt or K.taunt)(p, age)
	end
end

local cutToken = {} -- voice -> latest play, so an old cut never stops a newer footfall
local function makeVoice(def, root)
	local s = Instance.new("Sound")
	s.SoundId = "rbxassetid://" .. def.Id
	s.RollOffMode = Enum.RollOffMode.InverseTapered
	s.RollOffMinDistance = 18
	s.RollOffMaxDistance = NEAR
	s.Volume = 0
	s.Looped = def.Loop == true
	s.Parent = root
	return s
end
local function makeAudio(st)
	local prof = K.AUDIO[st.Key]
	local root = st.Model.PrimaryPart
	if not prof or not root then return end
	local a = {Prof = prof, Pool = {}, Count = 0}
	local function voices(name)
		if a.Pool[name] then return end
		local def = K.SOUNDS[name]
		a.Pool[name] = {Def = def, Voices = {makeVoice(def, root), makeVoice(def, root)}, I = 1}
	end
	if prof.Steps then
		voices(prof.Steps[1])
		voices(prof.Steps[2])
	end
	if prof.Layer then
		voices(prof.Layer[1])
	end
	if prof.Loop then
		a.Loop = makeVoice(K.SOUNDS[prof.Loop[1]], root)
	end
	st.Audio = a
end
local function cut(entry, pitch, volume)
	local v = entry.Voices[entry.I]
	entry.I = entry.I % #entry.Voices + 1
	local def = entry.Def
	v.PlaybackSpeed = pitch * (0.95 + math.random() * 0.1)
	v.Volume = def.Vol * volume
	v.TimePosition = def.Off or 0
	v:Play()
	local token = (cutToken[v] or 0) + 1
	cutToken[v] = token
	local len = (def.Len or 0.6) / v.PlaybackSpeed
	task.delay(len, function()
		if cutToken[v] == token then
			v:Stop()
		end
	end)
end
local function footfall(st, loud)
	local a = st.Audio
	if not a or not a.Prof.Steps then return end
	a.Count += 1
	local prof = a.Prof
	cut(a.Pool[prof.Steps[a.Count % 2 + 1]], prof.Pitch, prof.Vol * loud)
	local layer = prof.Layer
	if layer and a.Count % layer[4] == 0 then
		cut(a.Pool[layer[1]], layer[2], layer[3] * loud)
	end
end
local function silence(st)
	local a = st.Audio
	if a and a.Loop and a.Loop.IsPlaying then
		a.Loop:Stop()
	end
end

local function makeFx(st)
	st.FxObjects = {}
	st.Fx = {}
	local list = K.FX[st.Key]
	if not list then return end
	local model, s = st.Model, st.Rig.Scale
	for _, f in ipairs(list) do
		if f.Kind == "fin" then
			makeFin(st, s)
		elseif f.Kind == "mound" then
			makeMound(st, s)
		elseif f.Kind == "wave" then
			-- on the floor under the guard, not on the body
			local holder = Instance.new("Part")
			holder.Name = "GuardWave"
			holder.Anchored = true
			holder.CanCollide = false
			holder.CanQuery = false
			holder.CanTouch = false
			holder.Transparency = 1
			holder.Size = V(1, 1, 1)
			holder.Parent = workspace
			local att = Instance.new("Attachment")
			att.Parent = holder
			table.insert(st.FxObjects, holder)
			st.Wave = holder
			for _, e in ipairs(KINDS.wave(att, s)) do
				table.insert(st.Fx, e)
			end
		else
			local part = model:FindFirstChild(f.Part, true)
			if part and part:IsA("BasePart") and KINDS[f.Kind] then
				local att = Instance.new("Attachment")
				att.Name = "Fx_" .. f.Kind
				att.Position = f.At * part.Size
				att.Parent = part
				table.insert(st.FxObjects, att)
				local emitters, light = KINDS[f.Kind](att, s)
				for _, e in ipairs(emitters) do
					table.insert(st.Fx, e)
				end
				if light then
					local l = Instance.new("PointLight")
					l.Color = light[1]
					l.Range = 3 * s -- a local glow on the equipment, never a floodlight on the whole body
					l.Brightness = 0
					l.Shadows = false
					l.Parent = att
					table.insert(st.Fx, {l, light[2], Light = true})
				end
			end
		end
	end
end

local function track(model)
	if guards[model] or not model:IsA("Model") then return end
	guards[model] = {Model = model, Phase = 0, MoveW = 0, ActW = 0, Speed = 0, Moving = false, Seed = math.random() * 100,
		Run = 1, Lean = 0, Pitch = 0, Look = V(), Out = {}}
end
local function untrack(model)
	local st = guards[model]
	if st and st.Rig then
		for _, j in pairs(st.Rig.J) do
			if j.Parent then
				j.Transform = CFrame.identity
			end
		end
	end
	if st and st.Audio then
		for _, e in pairs(st.Audio.Pool) do
			for _, v in ipairs(e.Voices) do
				v:Destroy()
			end
		end
		if st.Audio.Loop then
			st.Audio.Loop:Destroy()
		end
	end
	if st and st.FxObjects then
		for _, o in ipairs(st.FxObjects) do
			o:Destroy()
		end
	end
	if st and st.DrawVoice then
		st.DrawVoice:Destroy()
	end
	guards[model] = nil
end

-- the nearest player's root for each guard, refreshed a few times a second (heads turn toward people nearby)
local lookTargets = {}
task.spawn(function()
	while true do
		local roots = {}
		for _, plr in ipairs(Players:GetPlayers()) do
			local r = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
			if r then
				table.insert(roots, r)
			end
		end
		for model, st in pairs(guards) do
			local root = model.PrimaryPart
			local best, bestD = nil, (st.Rig and st.Rig.Scale or 4) * 5.2 * LOOK_RANGE
			if root then
				for _, r in ipairs(roots) do
					local d = (r.Position - root.Position).Magnitude
					if d < bestD then
						best, bestD = r, d
					end
				end
			end
			lookTargets[model] = best
		end
		task.wait(0.3)
	end
end)

-- first sight of a guard: its rig, sounds, wings, gear and effects
local function setup(st)
	local model = st.Model
	st.Rig = K.measure(model)
	st.Key = model:GetAttribute("Motion")
	if not st.Rig then
		return false
	end
	makeAudio(st)
	for _, d in ipairs(model:GetDescendants()) do
		if d:IsA("Motor6D") and (d.Name == "LeftWing" or d.Name == "RightWing") then
			st.Wings = st.Wings or {}
			st.Wings[d.Name] = d
		end
	end
	local m = K.MOTION[st.Key]
	local carry = m and m.Carry and model:FindFirstChild(m.Carry.Joint, true)
	if carry and carry:GetAttribute("Grip") then
		local arm = carry:GetAttribute("GripArm")
		for _, d in ipairs(model:GetDescendants()) do
			if d:IsA("Motor6D") and d.Part1 and d.Part1.Name == arm and d.Part0 == carry.Part0 then
				-- the grip was measured on the stored rig; the giant is scaled up (Scale), its offsets with it
				local grip = carry:GetAttribute("Grip")
				grip = (grip - grip.Position) + grip.Position * st.Rig.Scale
				st.Carry = {Joint = carry, Grip = grip, Shoulder = d, Key = d.Name == "RightShoulder" and "RS" or "LS",
					Def = m.Carry, W = m.Carry.Held == "idle" and 1 or 0}
			end
		end
		if st.Carry and m.Carry.Held == "chase" and K.SOUNDS.draw then
			st.DrawVoice = makeVoice(K.SOUNDS.draw, model.PrimaryPart)
		end
	end
	for _, d in ipairs(model:GetDescendants()) do
		if d:IsA("Motor6D") and d.Name == "Swish" then
			st.Swish = st.Swish or {}
			table.insert(st.Swish, d)
		end
	end
	makeFx(st)
	return true
end

-- one carried item between its slot (its joint at rest) and the hand: W = 0 in the slot, 1 in the hand
local function updateCarry(st, model, tr, dt)
	local c = st.Carry
	local def = c.Def
	local chasing = model:GetAttribute("Chasing") == true
	local taunting = st.ActName == "taunt" and st.ActAge
	local k = taunting and K.ease((st.ActAge - def.From) / (def.To - def.From)) or 1
	local want
	if def.Held == "chase" then
		want = chasing and k or 0
	else
		want = chasing and 1 - k or 1
	end
	local before = c.W
	-- during the taunt it follows the draw exactly; otherwise it eases (a chase ending puts it away smoothly)
	c.W = taunting and want or c.W + (want - c.W) * (1 - exp(-dt * 6))
	if st.DrawVoice and before < 0.5 and c.W >= 0.5 then
		st.DrawVoice.Volume = K.SOUNDS.draw.Vol
		st.DrawVoice.TimePosition = 0
		st.DrawVoice:Play()
	end
	local sh, j = c.Shoulder, c.Joint
	local armRel = sh.C0 * tr[c.Key] * sh.C1:Inverse()
	local held = j.C0:Inverse() * armRel * c.Grip * j.C1
	j.Transform = CFrame.identity:Lerp(held, c.W)
end

------------------------------------------------------------------
-- EVERY FRAME
------------------------------------------------------------------
local function step(dt)
	local cam = workspace.CurrentCamera
	local camPos = cam and cam.CFrame.Position or Vector3.zero
	local now = workspace:GetServerTimeNow()
	local t = os.clock()
	for model, st in pairs(guards) do
		local root = model.Parent and model.PrimaryPart
		if not root then
			continue
		end
		if (root.Position - camPos).Magnitude > NEAR then
			st.Far = true
			silence(st)
			for _, e in ipairs(st.Fx or {}) do
				if e.Light then
					e[1].Brightness = 0
				else
					e[1].Rate = 0
				end
			end
			if st.Fin then
				st.Fin.Part.Transparency = 1
				st.Fin.Spray.Rate, st.Fin.Bubbles.Rate = 0, 0
			end
			continue
		end
		if not st.Rig and not setup(st) then
			continue
		end
		local rig = st.Rig
		local m = K.MOTION[st.Key] or K.MOTION.brutus
		local body = rig.Scale * 5.2 -- one body length in studs

		-- how far they really moved since last frame (a teleport home or a stream-in doesn't count)
		local pos = root.Position
		local moved = st.Last and V(pos.X - st.Last.X, 0, pos.Z - st.Last.Z).Magnitude or 0
		st.Last = pos
		if st.Far or moved > 600 * dt + 20 then
			moved = 0
			st.Far = false
		end
		local speed = moved / max(dt, 1 / 240) / body
		local before = st.Speed
		st.Speed += (speed - st.Speed) * (1 - exp(-dt * 10))
		if st.Moving and st.Speed < MOVE_OFF then
			st.Moving = false
		elseif not st.Moving and st.Speed > MOVE_ON then
			st.Moving = true
		end
		st.MoveW += ((st.Moving and 1 or 0) - st.MoveW) * (1 - exp(-dt * 7))
		-- fast movers (the quick walk home) take longer, bigger strides rather than spinning their legs faster
		local runTarget = model:GetAttribute("Returning") and clamp(st.Speed / 6, 0.8, 1.2) or 1
		st.Run += (runTarget - st.Run) * (1 - exp(-dt * 3))
		local strideScale = clamp(st.Speed / 5, 1, 1.8)

		-- stride phase from distance travelled; a footfall each time it passes the foot's landing point
		local foot = m.Foot or pi / 2
		local beforePhase = st.Phase
		st.Phase += moved / (m.Stride * rig.Scale * strideScale) * 2 * pi
		if st.Moving and floor((st.Phase - foot) / pi) ~= floor((beforePhase - foot) / pi) then
			footfall(st, st.Run >= 0.8 and 1 or 0.7)
			model:SetAttribute("StepTick", (model:GetAttribute("StepTick") or 0) + 1)
		end

		-- turning and speeding up lean the body (into the turn, forward when accelerating)
		local look = root.CFrame.LookVector
		local yaw = atan2(-look.X, -look.Z)
		local yawRate = st.Yaw and ((yaw - st.Yaw + pi) % (2 * pi) - pi) / max(dt, 1 / 240) or 0
		st.Yaw = yaw
		st.Lean += (clamp(yawRate * st.Speed * 0.05, -0.2, 0.2) * st.MoveW - st.Lean) * (1 - exp(-dt * 6))
		local accel = (st.Speed - before) / max(dt, 1 / 240)
		st.Pitch += (clamp(-accel * 0.03, -0.12, 0.08) - st.Pitch) * (1 - exp(-dt * 5))

		local s = {T = t + st.Seed, Run = st.Run}
		local pose = K.newPose()
		m.Idle(pose, t + st.Seed)
		if st.MoveW > 0.01 then
			local gp = K.newPose()
			m.Gait(gp, st.Phase, s)
			K.blend(pose, gp, st.MoveW)
		end
		pose.Root += V(st.Pitch, 0, st.Lean)

		-- the head turns toward the nearest player (while standing; a chasing guard faces its target anyway)
		local target = lookTargets[model]
		local want = V()
		if target and target.Parent then
			local d = root.CFrame:PointToObjectSpace(target.Position)
			local flat = math.sqrt(d.X * d.X + d.Z * d.Z)
			want = V(clamp(atan2(d.Y - body * 0.6, flat) * 0.6, -0.3, 0.25), clamp(atan2(-d.X, -d.Z), -0.9, 0.9), 0)
		end
		st.Look += (want - st.Look) * (1 - exp(-dt * 3))

		-- one-off reactions take over the body, crossfading from one to the next
		local act = model:GetAttribute("Act")
		local at = model:GetAttribute("ActAt") or 0
		local fresh = false
		if ACT_LEN[act] and (act ~= st.ActName or at ~= st.ActAt) then
			if st.ActName and st.ActW > 0.05 then
				st.Prev, st.PrevAge, st.PrevT = st.ActName, st.ActAge, t
			end
			st.ActName, st.ActAt = act, at
			fresh = true
		end
		local on = false
		if st.ActName then
			if act == st.ActName and at == st.ActAt then
				st.ActAge = max(0, now - at)
				on = st.ActAge < ACT_LEN[act]
			end
			st.ActW += ((on and 1 or 0) - st.ActW) * (1 - exp(-dt * (on and 16 or 5)))
			if st.ActW > 0.005 then
				local ap = K.newPose()
				actPose(m, st.ActName, st.ActAge, ap)
				if st.Prev and t - st.PrevT < CROSSFADE then
					local pp = K.newPose()
					actPose(m, st.Prev, st.PrevAge, pp)
					K.blend(pp, ap, K.ease((t - st.PrevT) / CROSSFADE))
					ap = pp
				end
				K.blend(pose, ap, st.ActW)
			else
				st.ActName = nil
			end
		end
		pose.Neck += st.Look * (1 - st.MoveW) * (1 - st.ActW)

		local tr = K.solve(rig, pose, st.Out)
		for key, joint in pairs(rig.J) do
			joint.Transform = tr[key]
		end

		-- wings: Fold sweeps them back along the body, Wing beats them (+ = up)
		if st.Wings then
			local fold, wing = pose.Fold * 1.15, pose.Wing
			if st.Wings.LeftWing then
				st.Wings.LeftWing.Transform = CFrame.Angles(0, fold, -wing)
			end
			if st.Wings.RightWing then
				st.Wings.RightWing.Transform = CFrame.Angles(0, -fold, wing)
			end
		end

		if st.Carry then
			updateCarry(st, model, tr, dt)
		end

		-- effects run with movement, and while airborne (a hover, an ascent)
		local w = max(st.MoveW, clamp(pose.Off.Y / 1.5, 0, 1))
		if st.Swish then
			for _, j in ipairs(st.Swish) do
				j.Transform = CFrame.Angles(0.1 * sin(t * 1.3) + 0.3 * sin(t * 4) * w, 0, 0)
			end
		end
		for _, e in ipairs(st.Fx) do
			if e.Light then
				e[1].Brightness = e[2] * w
			else
				e[1].Rate = e[2] * w
			end
		end
		if st.Wave then
			st.Wave.CFrame = CFrame.new(pos.X, FLOOR_Y + 0.5, pos.Z)
		end
		local fin = st.Fin
		if fin then
			-- the fin stands out of the water as far as the body has gone under it
			local under = clamp((pose.Sink or 0) / 6, 0, 1)
			local h = fin.Part.Size.Y
			if under > 0.3 then
				local rise = (under - 0.3) / 0.7
				fin.Part.Transparency = 0
				if fin.Kind == "mound" then
					fin.Part.CFrame = CFrame.new(pos.X, FLOOR_Y - h / 2 + h * 0.55 * rise, pos.Z) * CFrame.Angles(0, yaw, 0)
					fin.Spray.Rate = 30 * w * rise
					fin.Bubbles.Rate = 14 * w * rise
				else
					fin.Part.CFrame = CFrame.new(pos.X, WATER_Y - h / 2 + h * 0.85 * rise, pos.Z) * CFrame.Angles(0, yaw, 0)
					fin.Spray.Rate = 22 * w * rise
					fin.Bubbles.Rate = 10 * w * rise
					if w > 0.3 and t >= fin.Next then
						fin.Next = t + 0.16
						ripple(pos - look * fin.Part.Size.Z * 0.3, rig.Scale * 1.0, 1.1)
					end
				end
			else
				fin.Part.Transparency = 1
				fin.Spray.Rate, fin.Bubbles.Rate = 0, 0
			end
			-- the surge: a burst of spray and a wide ring where it breaks the surface
			if fresh and act == "grab" then
				fin.Spray:Emit(35)
				fin.Bubbles:Emit(fin.Kind == "mound" and 25 or 15)
				if fin.Kind ~= "mound" then
					ripple(pos, rig.Scale * 2.4, 1.2)
					ripple(pos, rig.Scale * 1.3, 0.8)
				end
			end
		end

		-- the flying and floating ones hum while they move
		local a = st.Audio
		if a and a.Loop then
			local def, lp = K.SOUNDS[a.Prof.Loop[1]], a.Prof.Loop
			local vol = def.Vol * lp[3] * w * (st.Run >= 0.8 and 1 or 0.7)
			if vol > 0.01 then
				a.Loop.Volume = vol
				a.Loop.PlaybackSpeed = lp[2] * (0.9 + 0.2 * w)
				if not a.Loop.IsPlaying then
					a.Loop:Play()
				end
			elseif a.Loop.IsPlaying then
				a.Loop:Stop()
			end
		end
	end
end

task.spawn(function()
	local map = workspace:WaitForChild("StealAGameMap", 60)
	local folder = map and map:WaitForChild("Shopkeepers", 60)
	if not folder then return end
	for _, m in ipairs(folder:GetChildren()) do
		track(m)
	end
	folder.ChildAdded:Connect(track)
	folder.ChildRemoved:Connect(untrack)
	RunService.PreSimulation:Connect(step)
end)
