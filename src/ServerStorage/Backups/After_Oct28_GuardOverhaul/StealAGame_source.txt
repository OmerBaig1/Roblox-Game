--[[
	STEAL A GAME - MAIN SERVER SCRIPT
	The rules of the game. The pieces it uses:
	  Config  - every balance number (speed, income, install times, prices, trails)
	  Kit     - small helpers (parts, labels, prompts, number formatting, remotes)
	  World   - the sky island, giant trees, the vault hallway, the Trail Shop
	  Base    - building player bases (floors, stairs, cases, podium, treadmill)
	  Guards  - shopkeeper models

	How it plays:
	- Run down the Game Vault hallway, grab a game, and get it home before that
	  level's shopkeeper catches you. Deeper levels have rarer games and faster shopkeepers.
	- Every copy of a game earns a random amount (bounded by its level). You don't
	  find out how much until it finishes installing in one of your cases.
	- Installing takes longer for rarer and higher-earning games. It runs at full
	  speed while you're in the game and half speed while you're offline.
	- Hold up to 2 games. Pick games up from your cases to rearrange or swap them.
	- Upgrade your base: more cases, then a new floor, then more cases...
	- Train on your treadmill for speed. Buy trails at the Trail Shop.
]]

local function getRoot(player)
	local char = player.Character
	if not char then return nil end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return nil end
	return char:FindFirstChild("HumanoidRootPart")
end

local function heldList(player)
	held[player] = held[player] or {}
	return held[player]
end

local function stolenCount(player)
	local n = 0
	for _, e in ipairs(heldList(player)) do
		if e.Kind ~= "own" then
			n += 1
		end
	end
	return n
end

local function updateSpeed(player)
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	local effective = (player:GetAttribute("SpeedPoints") or C.START_SPEED) * (player:GetAttribute("SpeedMultiplier") or 1)
	local stats = player:FindFirstChild("leaderstats")
	if stats and stats:FindFirstChild("Speed") then
		stats.Speed.Value = math.floor(effective)
	end
	local speed = C.realSpeed(effective)
	-- "Slow Down" switch: walk at the normal starting speed
	if player:GetAttribute("SlowDown") then
		speed = C.REAL_BASE_SPEED
	end
	if stolenCount(player) > 0 then
		speed *= C.CARRY_MULTIPLIER
	end
	hum.WalkSpeed = speed
end

local function rateText(rate)

end

-- Games from deeper than you've reached yourself only earn part of their money.
-- You "reach" a level by stealing from it in the vault (plus a couple of levels of grace).
local function levelFactor(owner, rec)
	local data = owner and playerData[owner]
	local reach = math.min(C.NUM_LEVELS, (data and data.BestLevel or 1) + C.STEAL_LEVEL_GRACE)
	if rec.L <= reach then
		return 1
	end
	return math.clamp(C.LEVELS[reach].IncomeBase / C.LEVELS[rec.L].IncomeBase, 0.01, 1)
end

-- The share of their games' income a player earns while offline. Everyone gets C.OFFLINE_CASH_RATE for now; a
-- future verified purchase (checked here) can raise it to 1.
local function offlineCashRate(player)
	return C.OFFLINE_CASH_RATE
end

local function freeSlot(plot)

		local slot = plot.Slots[i]
		if slot and not slot.Item and not slot.Reserved then

		end
	end
	return nil
end

-- GAME COPIES (random hidden income, install time)
------------------------------------------------------------------
local function pickGame(level)
	local list = C.LEVELS[level].Games
	local total = 0
	for _, g in ipairs(list) do
		total += g.Chance
	end
	local roll = math.random() * total
	for _, g in ipairs(list) do
		roll -= g.Chance
		if roll <= 0 then
			return g.GI
		end
	end
	return #list
end

local function makeRecord(level, gi, income, installTime, progress, r)
	local L = C.LEVELS[level]
	local g = L and L.Games[gi]
	if not g then return nil end
	r = math.clamp(math.floor(tonumber(r) or 1), 1, #C.RARITIES)
	local info = C.RARITIES[r]
	local rec = {Id = HttpService:GenerateGUID(false), L = level, G = gi, R = r, Name = g.Name, Rarity = info.Name, Color = info.Color, Income = income, InstallTime = installTime, Progress = progress or 0}
	rec.Installed = rec.Progress >= rec.InstallTime
	return rec
end

-- Roll a rarity for a copy found at this level (rare ones are much harder at low levels)
local function rollRarity(level)
	local weights = C.rarityWeights(level)
	local total = 0
	for _, w in ipairs(weights) do
		total += w
	end
	local roll = math.random() * total
	for r, w in ipairs(weights) do
		roll -= w
		if w > 0 and roll <= 0 then
			return r
		end
	end
	return 1
end

local function newGame(level, gi, forcedRarity)
	local L = C.LEVELS[level]
	local r = forcedRarity or rollRarity(level)
	-- income comes from the level's band and the rarity only (C.gameIncome), so deeper always means richer
	local income = C.gameIncome(level, r)
	local chanceFactor = (40 / L.Games[gi].Chance) ^ 0.3
	local installTime = math.max(5, math.floor(L.InstallBase * chanceFactor * C.RARITY_INSTALL_MULT ^ (r - 1) + 0.5))
	return makeRecord(level, gi, income, installTime, 0, r)
end

-- A save from before the expansion (no V) stored games as (old level 1-50, game 1-4): look the game up by name
-- and give it its new (level, game). Its income, rarity and install progress are kept as they were.
local function migrateGame(g)
	local names = C.LEGACY[tonumber(g.L)]
	local name = names and names[tonumber(g.G)]
	local level = name and C.GAME_LEVEL[name]
	if not level then return nil end
	for gi, def in ipairs(C.LEVELS[level].Games) do
		if def.Name == name then
			g.L, g.G = level, gi
			return g
		end
	end
	return nil
end

-- mode: "vault" (on a pedestal), "case" (in one of your cases) or "held" (over your head)
local function labelText(rec, mode)
	-- In the vault (and while you're carrying it out) the rarity stays a secret
	if mode == "vault" then
		return rec.Name
	end
	local head = string.format('<font color="#%s">%s</font>\n%s', rec.Color:ToHex(), rec.Rarity, rec.Name)
	if mode == "case" and rec.Installed and (rec.Factor or 1) < 1 then
		return head .. string.format('\n<font color="#5aff78">+$%s/s</font> <font color="#ffb43c">\u{1F512} %d%%</font>', fmt(rec.Income * rec.Factor), math.max(1, math.floor(rec.Factor * 100)))
	end
	if rec.Installed then
		return head .. '\n<font color="#5aff78">+$' .. fmt(rec.Income) .. '/s</font>'
	end
	return head .. '\n<font color="#ffd23c">Installing...</font>'
end

-- Sparkles and auras that get flashier the rarer the area
local function emitter(parent, props)
	local pe = Instance.new("ParticleEmitter")
	pe.LightEmission = 1
	pe.LockedToPart = false
	for k, v in pairs(props) do
		pe[k] = v
	end
	pe.Parent = parent
	return pe
end

local function addAura(item, rec)
	local zi = rec.R or 1
	local light = item:FindFirstChildOfClass("PointLight")
	if light then
		-- kept soft: the case's white posts and glass sit right next to it and blow out under bloom
		light.Range = 7 + zi * 0.6
		light.Brightness = 0.45 + zi * 0.07
	end
	if zi < 3 then return end
	local colors
	if zi >= 10 then
		colors = ColorSequence.new({
			ColorSequenceKeypoint.new(0, Color3.fromHSV(0, 0.6, 1)), ColorSequenceKeypoint.new(0.25, Color3.fromHSV(0.15, 0.6, 1)),
			ColorSequenceKeypoint.new(0.5, Color3.fromHSV(0.4, 0.6, 1)), ColorSequenceKeypoint.new(0.75, Color3.fromHSV(0.6, 0.6, 1)),
			ColorSequenceKeypoint.new(1, Color3.fromHSV(0.8, 0.6, 1)),
		})
	else
		colors = ColorSequence.new(rec.Color, rec.Color:Lerp(Color3.new(1, 1, 1), 0.6))
	end
	local aura = item -- effects come off the game itself

	-- Rare+: twinkles
	emitter(aura, {Shape = Enum.ParticleEmitterShape.Box, Rate = 3 + zi, Color = colors,
		Size = NumberSequence.new({NumberSequenceKeypoint.new(0, 0), NumberSequenceKeypoint.new(0.5, 0.35 + zi * 0.03), NumberSequenceKeypoint.new(1, 0)}),
		Lifetime = NumberRange.new(0.6, 1.2), Speed = NumberRange.new(0.3, 1.2), SpreadAngle = Vector2.new(180, 180)})
	if zi >= 4 then
		-- Epic+: a glowing shell around the game
		emitter(aura, {Shape = Enum.ParticleEmitterShape.Sphere, ShapeStyle = Enum.ParticleEmitterShapeStyle.Surface, ShapeInOut = Enum.ParticleEmitterShapeInOut.Outward, Rate = 10 + zi * 3, Color = colors,
			Size = NumberSequence.new(0.35 + zi * 0.04, 0), Transparency = NumberSequence.new(0.2, 1),
			Lifetime = NumberRange.new(0.8, 1.3), Speed = NumberRange.new(1.2, 1.8), Drag = 2})
	end
	if zi >= 5 then
		-- Legendary+: wisps rising off it
		emitter(aura, {Shape = Enum.ParticleEmitterShape.Box, EmissionDirection = Enum.NormalId.Top, Rate = 5 + zi, Color = colors,
			Size = NumberSequence.new(0.5, 0), Transparency = NumberSequence.new(0.1, 1),
			Lifetime = NumberRange.new(1, 1.6), Speed = NumberRange.new(2, 4), Acceleration = Vector3.new(0, 2, 0)})
	end
	if zi >= 7 then
		-- Godly+: streaks of light bursting outward
		emitter(aura, {Shape = Enum.ParticleEmitterShape.Sphere, ShapeStyle = Enum.ParticleEmitterShapeStyle.Surface, ShapeInOut = Enum.ParticleEmitterShapeInOut.Outward,
			Orientation = Enum.ParticleOrientation.VelocityParallel, Squash = NumberSequence.new(2.5), Rate = 6 + zi, Color = colors,
			Size = NumberSequence.new(0.35, 0), Lifetime = NumberRange.new(0.35, 0.6), Speed = NumberRange.new(6, 9)})
	end
	if zi >= 9 then
		-- Celestial+: big star twinkles
		emitter(aura, {Shape = Enum.ParticleEmitterShape.Sphere, Rate = 2 + zi * 0.3, Color = ColorSequence.new(Color3.new(1, 1, 1)),
			Size = NumberSequence.new({NumberSequenceKeypoint.new(0, 0), NumberSequenceKeypoint.new(0.5, 1.4), NumberSequenceKeypoint.new(1, 0)}),
			Lifetime = NumberRange.new(0.5, 0.9), Speed = NumberRange.new(0, 0.3), Rotation = NumberRange.new(0, 360)})
	end
end

-- Every game is a boxed release for one of six made-up consoles: CasePackaging builds the whole case (its plastic,
-- the front cover, its own back cover and the spines) and decides its size
local CasePackaging = require(script.CasePackaging)
local function addPosters(item, rec, hidden)
	CasePackaging.dress(item, rec, hidden)
end

local function createItem(rec, hidden)
	local item = Instance.new("Part")
	item.Name = rec.Name
	item.Size = CasePackaging.size(rec.Name)
	item.Anchored = true
	item.CanCollide = false
	item.Color = CasePackaging.plastic(rec.Name)
	item.Material = Enum.Material.SmoothPlastic
	item.Reflectance = 0.08
	item:SetAttribute("Hidden", hidden == true)
	addPosters(item, rec, hidden)
	local label = Kit.billboard(item, 260, 96, 4.6, labelText(rec, "vault"), 80)
	label.Parent.Name = "Info"
	-- Install countdown (the client fades it in when you're close to the case)
	local timer = Kit.billboard(item, 300, 46, 5.4, "", 60)
	timer.TextColor3 = GOLD
	timer.Parent.Name = "InstallTimer"
	timer.Parent.Enabled = false
	timer.Parent:SetAttribute("Active", false)
	CollectionService:AddTag(timer.Parent, "InstallTimer")
	local light = Instance.new("PointLight")
	light.Color = hidden and Color3.new(1, 1, 1) or rec.Color
	light.Range = 7
	light.Brightness = 0.45
	light.Shadows = false
	light.Parent = item
	if not hidden then
		addAura(item, rec)
	end
	return item
end

-- A game's rarity is revealed the moment it reaches someone's base
local function revealItem(item, rec)
	if not item:GetAttribute("Hidden") then return end
	item:SetAttribute("Hidden", false)
	item.Color = CasePackaging.plastic(rec.Name)
	addPosters(item, rec, false)
	local light = item:FindFirstChildOfClass("PointLight")
	if light then
		light.Color = rec.Color
	end
	addAura(item, rec)
end

-- Burst of sparkles and a little hop when a game finishes installing
local function celebrate(item, rec, bounce)
	local burst = Instance.new("ParticleEmitter")
	burst.Color = ColorSequence.new(rec.Color, Color3.new(1, 1, 1))
	burst.LightEmission = 1
	burst.Size = NumberSequence.new({NumberSequenceKeypoint.new(0, 0.8), NumberSequenceKeypoint.new(1, 0)})
	burst.Lifetime = NumberRange.new(0.6, 1.2)
	burst.Speed = NumberRange.new(10, 18)
	burst.SpreadAngle = Vector2.new(180, 180)
	burst.Rate = 0
	burst.Acceleration = Vector3.new(0, -12, 0)
	burst.Parent = item
	burst:Emit(50 + (rec.R or 1) * 10)
	local flash = Instance.new("PointLight")
	flash.Color = rec.Color
	flash.Range = 14
	flash.Brightness = 2
	flash.Parent = item
	TweenService:Create(flash, TweenInfo.new(1.2), {Brightness = 0}):Play()
	if bounce and item.Anchored then
		local home = item.CFrame
		local up = TweenService:Create(item, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {CFrame = home * CFrame.new(0, 1.6, 0) * CFrame.Angles(0, math.pi, 0)})
		up:Play()
		up.Completed:Connect(function()
			if item.Parent and item.Anchored then
				TweenService:Create(item, TweenInfo.new(0.45, Enum.EasingStyle.Bounce), {CFrame = home}):Play()
			end
		end)
	end
	task.delay(2, function()
		burst:Destroy()
		flash:Destroy()
	end)
end

local function clearItemExtras(item)
	for _, child in ipairs(item:GetChildren()) do
		if child:IsA("WeldConstraint") or child:IsA("ProximityPrompt") then
			child:Destroy()
		end
	end
	item.Anchored = true
	item.Massless = false
end

------------------------------------------------------------------
-- HOLDING GAMES (up to 2) out in front with both arms, the cover facing forward. A second game sits against the
-- first. Every client poses the carrier's arms around them (StealAGameClient, CARRYING).
------------------------------------------------------------------
local function carryOffset(i)
	-- upright, leaning back a little into the arms; the top edge stays below the chin
	return CFrame.new(0, -0.15 + (i - 1) * 0.2, -1.55 - (i - 1) * 0.7) * CFrame.Angles(math.rad(-8), 0, 0)
end
local function syncHeld(player)
	local list = heldList(player)
	local root = getRoot(player)
	local names = {}
	for i, e in ipairs(list) do
		local item = e.Item
		clearItemExtras(item)
		if root then
			item.CFrame = root.CFrame * carryOffset(i)
			item.Massless = true
			local weld = Instance.new("WeldConstraint")
			weld.Part0 = root
			weld.Part1 = item
			weld.Parent = item
			item.Anchored = false
			item.Parent = workspace
		else
			item.Parent = holdStore
		end
		setItemLabel(item, e.Data, e.Kind == "vault" and "vault" or "held")
		table.insert(names, e.Data.Name)
	end
	player:SetAttribute("Held", #list)
	player:SetAttribute("HeldNames", table.concat(names, "|"))
	updateSpeed(player)
end

local function maxHeld(player)
	return player:GetAttribute("Pass_DoubleSteal") == true and C.MAX_HELD_PASS or C.MAX_HELD
end

local function canTake(player)
	local plot = playerPlots[player]
	if not plot then
		notify(player, "You don't have a base!", RED)
		return false
	end
	if plot.Runner == player then
		notify(player, "Get off the treadmill first!", RED)
		return false
	end
	local max = maxHeld(player)
	if #heldList(player) >= max then
		if max < C.MAX_HELD_PASS then
			notify(player, "Your hands are full! Get 2x Steal in the Shop to carry 2 games.", RED)
		else
			notify(player, "Your hands are full! (" .. max .. " games max)", RED)
		end
		return false
	end
	return true
end

-- what selling a game pays: shown on its Sell prompt and paid by onSell, so the two always agree (the prompt used
-- to leave out a stolen game's earning factor)
local function sellValue(rec)
	return math.floor(rec.Installed and rec.Income * (rec.Factor or 1) * C.SELL_MULTIPLIER
		or C.LEVELS[rec.L].IncomeBase * C.UNINSTALLED_SELL * math.min(1, (rec.Factor or 1) * 4))
end

local function refreshSlotPrompts(plot, slot)
	local ownerId = plot.Owner and plot.Owner.UserId or 0
	local rec = slot.Data
	for _, pr in ipairs({slot.MainPrompt, slot.SellPrompt}) do
		pr:SetAttribute("OwnerId", ownerId)
		pr:SetAttribute("HasGame", rec ~= nil)
		pr:SetAttribute("Reserved", slot.Reserved)
		-- for the prompt panels (StealAGameClient): which game, and what selling it pays
		pr:SetAttribute("GameName", rec and rec.Name or nil)
		pr.ObjectText = rec and rec.Name or ("Case " .. slot.Index)
	end
	if rec then
		local value = sellValue(rec)
		slot.SellPrompt:SetAttribute("SellValue", value)
		slot.SellPrompt:SetAttribute("Installed", rec.Installed == true)
		slot.SellPrompt.ActionText = "Sell for $" .. fmt(value)
	end
end

------------------------------------------------------------------
-- INDEX: games you've found (best rarity) and claimed
------------------------------------------------------------------
local function pushIndex(player)
	local data = playerData[player]
	if not data then return end
	local state = {}
	for name, r in pairs(data.Found) do
		local level = C.GAME_LEVEL[name]
		if level then
			local reward = C.indexReward(level, r)
			state[name] = {R = r, C = data.Claimed[name] == true, Cash = reward.Cash, Speed = reward.Speed}
		end
	end
	player:SetAttribute("IndexState", HttpService:JSONEncode(state))
end

local function markFound(player, rec)
	local data = player and playerData[player]
	if not data then return end
	if (data.Found[rec.Name] or 0) < rec.R then
		local isNew = data.Found[rec.Name] == nil
		data.Found[rec.Name] = rec.R
		pushIndex(player)
		if isNew and data.Loaded then
			notify(player, "New game in your Index: " .. rec.Name .. "! Claim your reward there.", GOLD)
		end
	end
end

local function placeInSlot(plot, slot, item, rec)
	if plot.Owner then
		markFound(plot.Owner, rec)
	end
	revealItem(item, rec)
	clearItemExtras(item)
	item.CanCollide = false
	item.Parent = slot.Model
	item.CFrame = slot.ItemCF
	slot.Item, slot.Data, slot.Reserved = item, rec, false
	setItemLabel(item, rec, "case")
	refreshSlotPrompts(plot, slot)
end

local function takeFromSlot(plot, slot)
	local item, rec = slot.Item, slot.Data
	slot.Item, slot.Data = nil, nil
	refreshSlotPrompts(plot, slot)
	return item, rec
end

------------------------------------------------------------------
-- BASE LOCK TIMER. It counts down by the server clock: refreshLock charges the time since it last ran (unless the
-- lock was paused meanwhile), so a pause or resume between the 1s ticks is exact and pausing never adds or loses time.
-- It pauses while the owner is being chased, or while they carry a game stolen from the Vault - from the moment the
-- server hands it to them (grabFromVault) until they bring it home (finalizeStolen) or lose it (dropStolen: caught,
-- slapped, died, dropped, left, fell). Walking into the Vault alone doesn't pause it. Every one of those calls
-- refreshLock straight away; the 1s tick catches the chases.
-- The owner's attributes drive their countdown: LockLeft (seconds left at LockStamp), LockStamp (server time) and
-- LockPaused ("theft", "chase" or false); their screen counts down from those by itself.
------------------------------------------------------------------
local function lockPauseReason(plot)
	local owner = plot.Owner
	if not owner or plot.LockLeft <= 0 then
		return false
	end
	for _, e in ipairs(heldList(owner)) do
		if e.Kind == "vault" then
			return "theft"
		end
	end
	for _, g in pairs(guards) do
		if g.Target == owner then
			return "chase"
		end
	end
	return false
end

local function refreshLock(plot)
	local now = workspace:GetServerTimeNow()
	if plot.LockLeft > 0 and not plot.LockPaused then
		plot.LockLeft = math.max(0, plot.LockLeft - (now - (plot.LockStamp or now)))
	end
	plot.LockStamp = now
	plot.LockPaused = lockPauseReason(plot)
	if plot.Owner then
		plot.Owner:SetAttribute("LockLeft", plot.LockLeft)
		plot.Owner:SetAttribute("LockStamp", now)
		plot.Owner:SetAttribute("LockPaused", plot.LockPaused)
	end
end
local function refreshLockFor(player)
	local plot = player and playerPlots[player]
	if plot then
		refreshLock(plot)
	end
end
-- a lock starting (the pad) or reset to full (the 9 Robux product); a theft pause carries on
local function startLock(plot)
	plot.LockLeft = C.LOCK_TIME
	plot.LockStamp = nil
	plot.LockPaused = false
	refreshLock(plot)
end

local grabFromVault

local function putOnSpot(spot, item, rec)
	clearItemExtras(item)
	item.CFrame = spot.Pedestal.CFrame * CFrame.new(0, 3.7, 0)
	item.Parent = world.GamesFolder
	spot.Item, spot.Data, spot.Carried = item, rec, false
	setItemLabel(item, rec, "vault")
	local prompt = Kit.prompt(item, "Steal", rec.Name, 0.8, 10)
	prompt.Triggered:Connect(function(player)
		grabFromVault(player, spot)
	end)
end

local function spawnSpot(spot)
	if spot.Item or spot.Carried then return end
	-- a different game on each of the level's slots whenever it has enough (every level has 6-8)
	local shown = {}
	for _, other in ipairs(spots) do
		if other ~= spot and other.Level == spot.Level and other.Data then
			shown[other.Data.G] = true
		end
	end
	local gi = pickGame(spot.Level)
	for _ = 1, 20 do
		if not shown[gi] then
			break
		end
		gi = pickGame(spot.Level)
	end
	local rec = newGame(spot.Level, gi)
	putOnSpot(spot, createItem(rec, true), rec)
end

grabFromVault = function(player, spot)
	if not spot.Item or not canTake(player) then return end
	if not getRoot(player) then return end
	local item, rec = spot.Item, spot.Data
	spot.Item, spot.Data, spot.Carried = nil, nil, true
	table.insert(heldList(player), {Item = item, Data = rec, Kind = "vault", Spot = spot})
	syncHeld(player)
	refreshLockFor(player) -- a Vault theft pauses your base lock until you're home
	notify(player, "You grabbed " .. rec.Name .. "! Get it home!", GOLD)
end

-- Send a stolen game back where it came from
local function returnEntry(entry)
	if entry.Kind == "vault" then
		putOnSpot(entry.Spot, entry.Item, entry.Data)
	elseif entry.Kind == "base" then
		local plot, slot = entry.FromPlot, entry.FromSlot
		slot.Reserved = false
		if plot.Owner and plot.Owner == entry.FromOwner and plot.Slots[slot.Index] == slot and not slot.Item then
			placeInSlot(plot, slot, entry.Item, entry.Data)
			notify(plot.Owner, "Your " .. entry.Data.Name .. " was returned!", GREEN)
		else
			entry.Item:Destroy()
			if plot.Slots[slot.Index] == slot then
				refreshSlotPrompts(plot, slot)
			end
		end
		-- this theft is over: the owner's alarm for it stops
		if entry.FromOwner and entry.FromOwner.Parent then
			alertEvent:FireClient(entry.FromOwner, "returned", nil, entry.Data.Name, entry.TheftId)
		end
	end
end

-- Drop every stolen game this player holds (caught, slapped, died, left)
local function dropStolen(player, onlyLevel)
	local list = heldList(player)
	local dropped = false
	for i = #list, 1, -1 do
		local e = list[i]
		if e.Kind ~= "own" and (not onlyLevel or (e.Kind == "vault" and e.Spot.Level == onlyLevel)) then
			table.remove(list, i)
			returnEntry(e)
			dropped = true
		end
	end
	if dropped then
		syncHeld(player)
		refreshLockFor(player)
	end
	return dropped
end

-- A stolen game reached its new home
local function finalizeStolen(player, entry)
	if entry.Kind == "vault" then
		entry.Spot.Carried = false
		-- The deepest vault level you've stolen from decides how much stolen games can earn you
		local data = playerData[player]
		if data and entry.Spot.Level > (data.BestLevel or 1) then
			data.BestLevel = entry.Spot.Level
			player:SetAttribute("BestLevel", data.BestLevel)
		end
		notify(player, "You got " .. entry.Data.Name .. "! It's installing now.", GREEN)
	elseif entry.Kind == "base" then
		entry.FromSlot.Reserved = false
		if entry.FromPlot.Slots[entry.FromSlot.Index] == entry.FromSlot then
			refreshSlotPrompts(entry.FromPlot, entry.FromSlot)
		end
		if entry.FromOwner and entry.FromOwner.Parent then
			notify(entry.FromOwner, player.DisplayName .. " stole your " .. entry.Data.Name .. "!", RED)
			alertEvent:FireClient(entry.FromOwner, "stolen", player.DisplayName, entry.Data.Name, entry.TheftId)
		end
		local data = playerData[player]
		local reach = (data and data.BestLevel or 1) + C.STEAL_LEVEL_GRACE
		if entry.Data.L > reach then
			notify(player, "You stole " .. entry.Data.Name .. "! It's from level " .. entry.Data.L .. ", so it only earns part of its money until you steal from level " .. (entry.Data.L - C.STEAL_LEVEL_GRACE) .. " yourself.", GOLD)
		else
			notify(player, "You stole " .. entry.Data.Name .. "!", GREEN)
		end
	end
	entry.Kind = "own"
	refreshLockFor(player) -- home with it: the lock runs again
end

------------------------------------------------------------------
-- CASE BUTTONS: pick up / place / swap (owner), steal (others), sell (owner)
------------------------------------------------------------------
local theftCount = 0 -- each theft's id, so the owner's alarm can tell which one ended
local function onCase(player, plot, slot)
	if plot.Slots[slot.Index] ~= slot then return end
	if plot.Owner == player then
		local list = heldList(player)
		if plot.Runner == player then
			notify(player, "Get off the treadmill first!", RED)
			return
		end
		if slot.Data then
			if #list < maxHeld(player) then
				local item, rec = takeFromSlot(plot, slot)
				table.insert(list, {Item = item, Data = rec, Kind = "own"})
				syncHeld(player)
				notify(player, "Picked up " .. rec.Name, BLUE)
				soundEvent:FireClient(player, "pickup") -- only here, once the game is really in your hands
			else
				local e = table.remove(list, 1)
				local item, rec = takeFromSlot(plot, slot)
				table.insert(list, {Item = item, Data = rec, Kind = "own"})
				if e.Kind ~= "own" then
					finalizeStolen(player, e)
				end
				placeInSlot(plot, slot, e.Item, e.Data)
				syncHeld(player)
				notify(player, "Swapped " .. e.Data.Name .. " in for " .. rec.Name, BLUE)
				soundEvent:FireClient(player, "pickup")
			end
		else
			if slot.Reserved then
				notify(player, "A thief has this case's game! Slap them!", RED)
				return
			end
			if #list == 0 then
				notify(player, "Hold a game to put it in this case", BLUE)
				return
			end
			local e = table.remove(list, 1)
			if e.Kind ~= "own" then
				finalizeStolen(player, e)
			end
			placeInSlot(plot, slot, e.Item, e.Data)
			syncHeld(player)
			notify(player, "Placed " .. e.Data.Name .. " in case " .. slot.Index, BLUE)
		end
		return
	end

	-- Someone else's base: steal it
	if not slot.Data then return end
	if plot.LockLeft > 0 then
		notify(player, "This base is locked!", RED)
		return
	end
	if not canTake(player) or not getRoot(player) then return end
	local item, rec = takeFromSlot(plot, slot)
	slot.Reserved = true
	refreshSlotPrompts(plot, slot)
	theftCount += 1
	table.insert(heldList(player), {Item = item, Data = rec, Kind = "base", FromPlot = plot, FromSlot = slot, FromOwner = plot.Owner, TheftId = theftCount})
	syncHeld(player)
	notify(player, "You grabbed " .. rec.Name .. "! Run back to your base!", GOLD)
	if plot.Owner then
		notify(plot.Owner, player.DisplayName .. " is stealing your " .. rec.Name .. "! Slap them!", RED)
		alertEvent:FireClient(plot.Owner, "stealing", player.DisplayName, rec.Name, theftCount)
	end
end

local function onSell(player, plot, slot)
	if plot.Owner ~= player or not slot.Data or plot.Slots[slot.Index] ~= slot then return end
	local rec = slot.Data
	local value = sellValue(rec)
	local item = takeFromSlot(plot, slot)
	item:Destroy()
	player.leaderstats.Cash.Value += value
	notify(player, "Sold " .. rec.Name .. " for $" .. fmt(value), GREEN)
	soundEvent:FireClient(player, "coin")
end

local caseCallbacks = {onCase = onCase, onSell = onSell}

-- Buy the next case (or floor). Called from the glowing upgrade spots in the base.
local refreshPlaceholders
local function doUpgrade(player, plot)
	if plot.Owner ~= player then
		notify(player, "That's not your base!", RED)
		return
	end
	local kind, cost = C.nextUpgrade(plot.Floors, plot.Cases, plot.Tier)
	if not kind then
		notify(player, "Your base is fully upgraded!", GOLD)
		return
	end
	local cash = player.leaderstats.Cash
	if cash.Value < cost then
		notify(player, "You need $" .. fmt(cost) .. " for that upgrade", RED)
		return
	end
	cash.Value -= cost
	soundEvent:FireClient(player, "buy")
	local data = playerData[player]
	data.Upgrades += 1
	local floors, cases, tier = C.baseLayout(data.Upgrades)
	if tier ~= plot.Tier then
		-- Revamp: take every game out, rebuild the bigger base, put them back
		local keep = {}
		for i = 1, plot.Cases do
			local slot = plot.Slots[i]
			if slot and slot.Item then
				local item, rec = takeFromSlot(plot, slot)
				item.Parent = nil
				table.insert(keep, {item, rec})
			end
		end
		Base.setTier(plot, tier)
		Base.setLayout(plot, floors, cases, caseCallbacks)
		setPlotOwnerPrompts(plot)
		for _, k in ipairs(keep) do
			local slot = freeSlot(plot)
			if slot then
				placeInSlot(plot, slot, k[1], k[2])
			else
				table.insert(heldList(player), {Item = k[1], Data = k[2], Kind = "own"})
			end
		end
		syncHeld(player)
		refreshPlaceholders(plot)
		notify(player, "Your base is now a " .. C.BASE_TIERS[tier].Name .. "! More room for cases!", GOLD)
		return
	end
	Base.setLayout(plot, floors, cases, caseCallbacks)
	setPlotOwnerPrompts(plot)
	refreshPlaceholders(plot)
	if kind == "floor" then
		notify(player, "Floor " .. floors .. " built! Take the stairs on the side.", GOLD)
	else
		notify(player, "Case " .. cases .. " added!", GOLD)
	end
end

refreshPlaceholders = function(plot)
	local kind, cost = C.nextUpgrade(plot.Floors, plot.Cases, plot.Tier)
	Base.setPlaceholders(plot, kind, cost and ("$" .. fmt(cost)) or "", doUpgrade)
end
local updateUpgradeLabel = refreshPlaceholders

-- The machine's running sounds (made by TreadModels, silent until used): fade in when someone gets on, out when
-- they get off. A rebuilt machine (an upgrade mid-run) brings new sounds, started by rebuildTreadmill.
local function setTreadSound(plot, on)
	if not plot.TreadModel then return end
	for _, s in ipairs(plot.TreadModel:GetDescendants()) do
		if s:IsA("Sound") and s:GetAttribute("Volume") then
			if on then
				if not s.IsPlaying then
					s.Volume = 0
					s:Play()
				end
				TweenService:Create(s, TweenInfo.new(0.6), {Volume = s:GetAttribute("Volume")}):Play()
			else
				local tw = TweenService:Create(s, TweenInfo.new(0.35), {Volume = 0})
				tw.Completed:Connect(function(state)
					if state == Enum.PlaybackState.Completed and not plot.Runner then
						s:Stop()
					end
				end)
				tw:Play()
			end
		end
	end
end

local function rebuildTreadmill(plot)
	local owner = plot.Owner
	local level = owner and (owner:GetAttribute("TreadLevel") or 0) or 0
	local name = TreadModels.get(level).Name
	-- the prompts show only for the owner (client); the Triggered handlers below check it again
	plot.RunPrompt:SetAttribute("OwnerId", owner and owner.UserId or 0)
	plot.TreadUpgradePrompt:SetAttribute("OwnerId", owner and owner.UserId or 0)
	-- for the panel's upgrade row (display only: the Triggered handler below checks cash and level itself)
	plot.TreadUpgradePrompt:SetAttribute("Maxed", level >= C.TREAD_MAX_LEVEL)
	plot.TreadUpgradePrompt:SetAttribute("Cost", level < C.TREAD_MAX_LEVEL and C.treadCost(level) or 0)
	Base.rebuildTreadmill(plot, level, string.format("%s TIER \u{00B7} %s  Lv %d\n+%s \u{1F45F}/s", string.upper(TreadModels.tier(level).Name), string.upper(name), level, rateText(C.treadGain(level))))
	if plot.Runner then
		setTreadSound(plot, true)
	end
	if not owner then
		plot.TreadUpgradePrompt.ActionText = "Upgrade Treadmill"
	elseif level >= C.TREAD_MAX_LEVEL then
		plot.TreadUpgradePrompt.ActionText = "Treadmill maxed!"
	else
		plot.TreadUpgradePrompt.ActionText = "Upgrade ($" .. fmt(C.treadCost(level)) .. ")"
	end
end

local function stopTreadmill(plot)
	local runner = plot.Runner
	if not runner then return end
	plot.Runner = nil
	setTreadSound(plot, false)
	plot.RunPrompt.ActionText = "Run on Treadmill"
	plot.RunPrompt:SetAttribute("Running", false)
	local root = runner.Character and runner.Character:FindFirstChild("HumanoidRootPart")
	if root then
		root.Anchored = false
		root.CFrame = plot.TreadCF * CFrame.new(0, 4, 1) -- just behind the pad
	end
	if runner.Parent then
		treadmillEvent:FireClient(runner, false)
	end
end

local function startTreadmill(plot, player)
	local char = player.Character
	local root = getRoot(player)
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not root or not hum then return end
	local height = hum.RigType == Enum.HumanoidRigType.R6 and 3 or (hum.HipHeight + root.Size.Y / 2)
	plot.Runner = player
	plot.TreadTime = 0
	plot.RunPrompt.ActionText = "Stop Running"
	plot.RunPrompt:SetAttribute("Running", true)
	root.Anchored = true
	root.CFrame = plot.TreadCF * CFrame.new(0, Base.BELT_TOP + height, -10)
	treadmillEvent:FireClient(player, true)
	setTreadSound(plot, true)
end

		refreshSlotPrompts(plot, slot)
	end
end

			plot.Bank -= amount
			notify(player, "Collected $" .. fmt(amount) .. "!", GREEN)
			soundEvent:FireClient(player, "coin")
			-- the pad shows $0 (in gold) for a moment so the collection reads. Only the label waits: the bank keeps
			-- filling underneath, and a newer collection just moves the same deadline
			plot.PadColor = plot.PadColor or plot.PadLabel.TextColor3
			plot.PadShowZeroUntil = os.clock() + 1.2
			plot.PadLabel.Text = "$0\n+$" .. fmt(plot.LastIncome or 0) .. "/s"
			plot.PadLabel.TextColor3 = Color3.fromRGB(255, 215, 80)
		end
	end)

	plot.RunPrompt.Triggered:Connect(function(player)
		if plot.Owner ~= player then
			notify(player, "That's not your treadmill!", RED)
		elseif plot.Runner then
			stopTreadmill(plot)
		elseif #heldList(player) > 0 then
			notify(player, "Put down your games first!", RED)
		else
			startTreadmill(plot, player)
		end
	end)

	plot.TreadUpgradePrompt.Triggered:Connect(function(player)
		if plot.Owner ~= player then
			notify(player, "That's not your treadmill!", RED)
			return
		end
		local level = player:GetAttribute("TreadLevel") or 0
		if level >= C.TREAD_MAX_LEVEL then
			notify(player, "Your treadmill is already maxed!")
			return
		end
		local cost = C.treadCost(level)
		local cash = player.leaderstats.Cash
		if cash.Value < cost then
			notify(player, "You need $" .. fmt(cost) .. " to upgrade the treadmill", RED)
			return
		end
		cash.Value -= cost
		player:SetAttribute("TreadLevel", level + 1)
		rebuildTreadmill(plot)
		notify(player, "Unlocked the " .. TreadModels.get(level + 1).Name .. " treadmill! Now +" .. rateText(C.treadGain(level + 1)) .. " speed every second", BLUE)
	end)

	plot.LockLeft = 0
	plot.LockButton.Touched:Connect(function(hit)
		local player = Players:GetPlayerFromCharacter(hit.Parent)
		-- relocking is instant: the unlock button is up on the wall, out of reach of the pad
		if player and player == plot.Owner and plot.LockLeft <= 0 then
			startLock(plot)
			plot.Barrier.Transparency = 0.45
			if plot.UnlockPrompt then
				plot.UnlockPrompt.MaxActivationDistance = 8
			end
			notify(player, "Base locked for " .. Kit.fmtTime(C.LOCK_TIME) .. "! (The timer pauses while you're being chased)", GREEN)
			soundEvent:FireClient(player, "lock")
		end
	end)
	-- Hold the UNLOCK button on the wall to open your base early
	if plot.UnlockPrompt then
		plot.UnlockPrompt.Triggered:Connect(function(player)
			if player ~= plot.Owner or plot.LockLeft <= 0 then return end
			plot.LockLeft = 0
			refreshLock(plot)
			plot.UnlockPrompt.MaxActivationDistance = 0
			for _, b in ipairs(Base.barriers(plot)) do
				b.Transparency = 1
				b.CanCollide = false
			end
			plot.LockLabel.Text = "LOCK BASE"
			notify(player, "Base unlocked!", BLUE)
			soundEvent:FireClient(player, "unlock")
			-- the button flashes as a "done!"
			local btn = plot.UnlockButton
			if btn then
				btn.Color = Color3.fromRGB(255, 255, 255)
				task.delay(0.35, function()
					btn.Color = Color3.fromRGB(70, 220, 110)
				end)
			end
		end)
	end

	table.insert(plots, plot)
end

		g.ReturnSpeed = math.clamp(g.RealSpeed * C.GUARD_RETURN_FACTOR, C.GUARD_RETURN_MIN, C.GUARD_RETURN_MAX) * (Guards.ROSTER[level].Pace or 1)
		g.IdleText = guardLabelText(g)
		-- float the name above the tallest hat/helmet piece (a Toy Soldier's shako and plume covered it)
		local top = head and head.Size.Y / 2 or 2
		if head then
			for _, p in ipairs(model:GetDescendants()) do
				local d = p:IsA("BasePart") and p.Position - head.Position
				if d and Vector3.new(d.X, 0, d.Z).Magnitude < head.Size.X then
					-- largest side, since hats like the shako are cylinders turned upright
					top = math.max(top, d.Y + math.max(p.Size.X, p.Size.Y, p.Size.Z) / 2)
				end
			end
		end
		g.Label = Kit.billboard(head or root, 420, 96, top + 2.2, g.IdleText, 160)
		g.Label.TextWrapped = false -- one line at every distance (it used to drop the last word when you got close)
		guards[level] = g
	end)
end

local function holdsFromLevel(player, level)
	for _, e in ipairs(heldList(player)) do
		if e.Kind == "vault" and e.Spot.Level == level and not e.Escaped then
			return true
		end
	end
	return false
end

local function setGuardTarget(g, player, now)
	if g.Target == player then return end
	if g.Target and g.Target.Parent then
		chaseEvent:FireClient(g.Target, false)
	end
	g.Target = player
	if g.Grab then
		-- a lunge in progress ends with the chase
		g.Grab = nil
		g.Actions.Act("")
	end
	-- the players' games pick the chase gait or the walk home from these
	g.Model:SetAttribute("Chasing", player ~= nil)
	if player then
		g.ReturnStart = nil
		g.Model:SetAttribute("Returning", false)
		g.ChaseAt = now + C.GUARD_REACTION
		g.ChaseStart = now
		g.SpeedNow = 0
		chaseEvent:FireClient(player, true, g.Title)
		-- the giant turns to face you (updateGuard, during its reaction time) and points before charging
		g.Actions.Taunt()
	end
end

local function catchPlayer(g, player)
	local tr = getRoot(player)
	setGuardTarget(g, nil, os.clock())
	dropStolen(player, g.Level)
	notify(player, "The " .. g.Title .. " caught you! The game was returned.", RED)
	caughtEvent:FireClient(player, g.Title)
	if tr then
		local away = tr.Position - g.Root.Position
		away = Vector3.new(away.X, 0, away.Z)
		if away.Magnitude < 0.1 then
			away = g.Root.CFrame.LookVector
		end
		-- knocked away from the guard's hands, which are right on you: no teleport, the grab only lands in reach
		knockEvent:FireClient(player, away.Unit * 50 + Vector3.new(0, 30, 0))
	end
	g.Actions.Laugh()
	g.PauseUntil = os.clock() + C.GUARD_LAUGH_TIME
end

-- Recovery only (see Config: stuck, out too long, or far away with nobody to see): back at the post in a puff
local function sendHome(g, now)
	g.Model:PivotTo(CFrame.lookAt(g.Post, g.Post + Vector3.new(0, 0, -1)))
	g.Humanoid:MoveTo(g.Post)
	g.ReturnStart = nil
	g.SpeedNow = 0
	g.CheckPos, g.CheckTime = g.Post, now
end

local function insideHall(p)
	return Vector3.new(math.clamp(p.X, -C.ROOM_W / 2 + 4, C.ROOM_W / 2 - 4), p.Y, math.clamp(p.Z, C.HALL_START_Z + 4, world.HallEnd - 4))
end

-- A REAL escape: you are out past the entrance (or outside the walls), judged by where
-- you stand on the ground plan - height is ignored, so a big jump never counts.
local function outOfVault(p)
	return p.Z < C.HALL_START_Z - 4 or math.abs(p.X) > C.ROOM_W / 2 + 8 or p.Z > world.HallEnd + 8
end

local function approach(x, target, step)
	if x < target then
		return math.min(target, x + step)
	end
	return math.max(target, x - step)
end

-- Turns a standing guard toward a point, no faster than GUARD_TURN_RATE (no snapping round). True once facing it.
local function turnToward(g, point, dt)
	local root = g.Root
	local p = root.Position
	local want = Vector3.new(point.X - p.X, 0, point.Z - p.Z)
	if want.Magnitude < 0.5 then
		return true
	end
	local look = root.CFrame.LookVector
	local cur = math.atan2(-look.X, -look.Z)
	local d = (math.atan2(-want.X, -want.Z) - cur + math.pi) % (2 * math.pi) - math.pi
	local turn = math.clamp(d, -C.GUARD_TURN_RATE * dt, C.GUARD_TURN_RATE * dt)
	root.CFrame = CFrame.new(p) * CFrame.Angles(0, cur + turn, 0)
	return math.abs(d - turn) < 0.03
end

local function seenByAnyone(pos, dist)
	for _, p in ipairs(Players:GetPlayers()) do
		local r = getRoot(p)
		if r and (r.Position - pos).Magnitude < dist then
			return true
		end
	end
	return false
end

-- The catch check, decided here on the server (GuardCatch explains the why): within the guard's reach both where
-- you are now and on your own screen, feet at the same height, and nothing solid between you
local catchRay = RaycastParams.new()
catchRay.FilterType = Enum.RaycastFilterType.Exclude
catchRay.RespectCanCollide = true
local function canCatch(g, player, tRoot, slack)
	local root = g.Root
	local gp, tp = root.Position, tRoot.Position
	-- compare feet, since a giant's middle is far above yours
	if math.abs((tp.Y - 3) - (gp.Y - g.StandY)) >= 6 then
		return false
	end
	local ping = player:GetNetworkPing()
	local shown = GuardCatch.shown(g.Hist, os.clock() - GuardCatch.delay(C, ping)) or gp
	if not GuardCatch.inReach(C, g.Reach, gp, shown, tp, tRoot.AssemblyLinearVelocity, ping, slack) then
		return false
	end
	local ignore = {world.GuardsFolder}
	for _, p in ipairs(Players:GetPlayers()) do
		if p.Character then
			table.insert(ignore, p.Character)
		end
	end
	catchRay.FilterDescendantsInstances = ignore
	local from = Vector3.new(gp.X, tp.Y, gp.Z)
	return workspace:Raycast(from, tp - from, catchRay) == nil
end

local function updateGuard(g, now)
	local root, hum = g.Root, g.Humanoid
	if not root.Parent then return end
	-- a puff of dust with every giant stride
	local pos = root.Position
	g.StepDist += Vector3.new(pos.X - g.LastPos.X, 0, pos.Z - g.LastPos.Z).Magnitude
	g.LastPos = pos
	if g.StepDist >= 2.2 * g.Scale then
		g.StepDist = 0
		g.Actions.Stomp()
	end
	-- where it has been over the last moment: what players' screens are still showing (GuardCatch.shown)
	table.insert(g.Hist, {T = now, P = pos})
	while #g.Hist > 2 and now - g.Hist[1].T > 0.8 do
		table.remove(g.Hist, 1)
	end
	if now < g.PauseUntil then
		hum:MoveTo(root.Position)
		-- standing still on purpose (laughing, recovering from a missed grab) is not "stuck"
		g.CheckPos, g.CheckTime = root.Position, now
		return
	end
	local t = g.Target
	if t then
		local tr = t.Parent and getRoot(t)
		local valid = tr and holdsFromLevel(t, g.Level)
		-- must stay out for a moment, so touching a boundary at high speed doesn't count
		if valid and outOfVault(tr.Position) then
			g.OutSince = g.OutSince or now
		else
			g.OutSince = nil
		end
		if valid and g.OutSince and now - g.OutSince >= C.ESCAPE_CONFIRM_TIME then
			g.OutSince = nil
			for _, e in ipairs(heldList(t)) do
				if e.Kind == "vault" and e.Spot.Level == g.Level then
					e.Escaped = true
				end
			end
			notify(t, "You escaped the Vault! The " .. g.Title .. " stopped chasing.", GREEN)
			valid = false
		end
		if valid and now - g.ChaseStart > C.GUARD_GIVE_UP then
			for _, e in ipairs(heldList(t)) do
				if e.Kind == "vault" and e.Spot.Level == g.Level then
					e.Escaped = true
				end
			end
			notify(t, "The " .. g.Title .. " gave up!", GREEN)
			valid = false
		end
		if not valid then
			setGuardTarget(g, nil, now)
			t = nil
		end
	end
	if not t then
		for p in pairs(held) do
			if p.Parent and holdsFromLevel(p, g.Level) and getRoot(p) then
				setGuardTarget(g, p, now)
				t = p
				break
			end
		end
	end

	if t then
		g.Label.Text = '<font color="#ff4040">!!! ' .. string.upper(g.Name) .. ' !!!</font>'
		local tRoot = getRoot(t)
		if now < g.ChaseAt then
			-- reaction time: stands, turns to face you and points
			hum:MoveTo(root.Position)
			turnToward(g, tRoot.Position, 0.05)
			return
		end
		-- slower than the level's recommended speed: the shopkeeper always out-runs you.
		-- Judged on speed points, not WalkSpeed: WalkSpeed is a float32 (a player exactly at the
		-- recommended speed read a hair under it) and is capped at MAX_REAL_SPEED.
		local tHum = t.Character and t.Character:FindFirstChildOfClass("Humanoid")
		local speed = g.RealSpeed
		local points = (t:GetAttribute("SpeedPoints") or C.START_SPEED) * (t:GetAttribute("SpeedMultiplier") or 1)
		if tHum and points < g.MustBeat then
			speed = math.max(speed, tHum.WalkSpeed * 1.25)
		end
		-- it stops just in front of someone standing still; someone fleeing pulls away from any point it aims short of,
		-- so the faster they run the more it aims right at them (measured: at 80 studs/s any standoff left it trailing
		-- out of reach). The lunge adds a burst of speed while the hands go out
		local tVel = tRoot.AssemblyLinearVelocity
		local standoff = g.Standoff * math.clamp(1 - Vector3.new(tVel.X, 0, tVel.Z).Magnitude / 60, 0, 1)
		if g.Grab then
			speed *= C.GUARD_LUNGE_BOOST
			standoff -= 1
		end
		-- momentum: from standing to full speed over GUARD_ACCEL_TIME
		g.SpeedNow = approach(g.SpeedNow or 0, speed, speed * 0.05 / C.GUARD_ACCEL_TIME)
		hum.WalkSpeed = g.SpeedNow
		-- run at where you'll be by the time this movement reaches your screen, stopping just in front of you (the
		-- giant's body never runs through you)
		local aim = GuardCatch.aim(C, tRoot.Position, tRoot.AssemblyLinearVelocity, t:GetNetworkPing())
		local to = Vector3.new(aim.X - root.Position.X, 0, aim.Z - root.Position.Z)
		if to.Magnitude > standoff then
			hum:MoveTo(insideHall(aim - to.Unit * standoff))
		else
			hum:MoveTo(root.Position)
			turnToward(g, tRoot.Position, 0.05)
		end
		if g.Grab then
			-- mid-lunge: the hands arrive GRAB_TIME after it began. Still in reach = caught; slipped away = a stumble
			if now - g.Grab >= C.GUARD_GRAB_TIME then
				g.Grab = nil
				if canCatch(g, t, tRoot, C.GUARD_GRAB_SLACK) then
					catchPlayer(g, t)
				else
					-- a stumble: it loses speed and can't grab again until it has recovered, but keeps coming
					g.Actions.Act("whiff")
					g.NoGrabUntil = now + C.GUARD_WHIFF_TIME
					g.SpeedNow = speed * 0.4
				end
			end
		elseif now >= (g.NoGrabUntil or 0) and canCatch(g, t, tRoot, 1) then
			g.Grab = now
			g.Actions.Act("grab")
		end
	else
		g.Label.Text = g.IdleText
		local flat = Vector3.new(g.Post.X - root.Position.X, 0, g.Post.Z - root.Position.Z)
		local dist = flat.Magnitude
		if dist > 2 then
			g.Settled = false
			g.ReturnStart = g.ReturnStart or now
			g.Model:SetAttribute("Returning", true)
			-- a deliberate walk home at its own pace (slowing down out of the chase); far outside its own level it
			-- hurries, so it covers the rest of the way in about GUARD_FAR_TIME
			local speed = g.ReturnSpeed
			if dist > C.GUARD_FAR_DIST then
				speed = math.max(speed, (dist - C.GUARD_FAR_DIST) / C.GUARD_FAR_TIME)
			end
			g.SpeedNow = approach(g.SpeedNow or 0, speed, math.max(speed, g.SpeedNow or 0) * 0.05 / C.GUARD_ACCEL_TIME)
			hum.WalkSpeed = g.SpeedNow
			hum:MoveTo(g.Post)
			-- recovery (Config): stuck even after a jump, out too long, or far from home with nobody to see it
			if now - g.CheckTime > C.GUARD_STUCK_TIME then
				local stuck = (root.Position - g.CheckPos).Magnitude < 1.5
				g.CheckPos, g.CheckTime = root.Position, now
				if stuck and not g.TriedJump then
					g.TriedJump = true
					hum.Jump = true
					stuck = false
				end
				if stuck or now - g.ReturnStart > C.GUARD_RETURN_TIMEOUT
					or (dist > C.GUARD_FAR_DIST and not seenByAnyone(root.Position, C.GUARD_UNSEEN_DIST)) then
					g.Actions.Poof()
					sendHome(g, now)
				end
			end
		elseif not g.Settled then
			if g.Model:GetAttribute("Returning") then
				-- arrived: stop walking completely
				g.ReturnStart = nil
				g.TriedJump = false
				g.SpeedNow = 0
				g.Model:SetAttribute("Returning", false)
				hum:Move(Vector3.zero)
				hum:MoveTo(root.Position)
				hum.WalkSpeed = C.GUARD_IDLE_REAL_SPEED
			end
			-- then turn back to face the entrance at a natural pace
			if turnToward(g, root.Position + Vector3.new(0, 0, -10), 0.05) then
				g.Settled = true
			end
		end
	end
end

local function ownedTrails(player)
	local ok, list = pcall(function()
		return HttpService:JSONDecode(player:GetAttribute("TrailsOwned") or "[]")
	end)
	return ok and type(list) == "table" and list or {}
end

local function ownsTrail(player, idx)
	for _, i in ipairs(ownedTrails(player)) do
		if i == idx then
			return true
		end
	end
	return false
end

-- How each trail looks. Everything is built here so everyone sees it; the rainbow
-- colour flow and the lightning bolts are animated on each player's own screen
-- (StealAGameClient), which keeps the server from sending updates every frame.
local TEX_SPARKLE = "rbxasset://textures/particles/sparkles_main.dds"
local TEX_FIRE = "rbxasset://textures/particles/fire_main.dds"
local TEX_SMOKE = "rbxasset://textures/particles/smoke_main.dds"

local function applyTrail(player)
	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return end
	for _, child in ipairs(root:GetChildren()) do
		if child.Name == "StealTrail" then
			child:Destroy()
		end
	end
	root:SetAttribute("TrailFx", nil)
	root:SetAttribute("TrailColor", nil)
	root:SetAttribute("TrailRainbow", nil)
	root:SetAttribute("TrailBubbles", nil)
	local idx = player:GetAttribute("TrailEquipped") or 0
	local t = C.TRAILS[idx]
	if not t then return end
	local rgb = Kit.rgb
	local power = idx / #C.TRAILS -- 0 to 1: better trails get more of everything
	local function kp(time, value)
		return NumberSequenceKeypoint.new(time, value)
	end

	local function attach(y)
		local a = Instance.new("Attachment")
		a.Name = "StealTrail"
		a.Position = Vector3.new(0, y, 0)
		a.Parent = root
		return a
	end
	local top, bottom = attach(1.1), attach(-1.1)
	local keys = {}
	for i, c in ipairs(t.Colors) do
		keys[i] = ColorSequenceKeypoint.new((i - 1) / math.max(1, #t.Colors - 1), c)
	end
	if #keys == 1 then
		keys[2] = ColorSequenceKeypoint.new(1, t.Colors[1])
	end
	local colors = ColorSequence.new(keys)
	local function tagRainbow(obj)
		if t.Animated then
			CollectionService:AddTag(obj, "RainbowTrail")
		end
	end
	local function trail(props)
		local tr = Instance.new("Trail")
		tr.Name = "StealTrail"
		tr.Attachment0, tr.Attachment1 = top, bottom
		tr.FaceCamera = true
		tr.MinLength = 0.05
		tr.Color = colors
		for k, v in pairs(props) do
			tr[k] = v
		end
		tr.Parent = root
		tagRainbow(tr)
		return tr
	end
	-- near-white trails glare under bloom, so they glow about half as much
	local lum = 0
	for _, c in ipairs(t.Colors) do
		lum += (c.R * 0.3 + c.G * 0.59 + c.B * 0.11) / #t.Colors
	end
	local glow = lum > 0.75 and 0.5 or 1
	local function emitter(props)
		local pe = Instance.new("ParticleEmitter")
		pe.Name = "StealTrail"
		pe.LightEmission = 0.6 * glow
		pe.SpreadAngle = Vector2.new(180, 180)
		for k, v in pairs(props) do
			pe[k] = v
		end
		pe.Parent = root
		return pe
	end
	local function light(color, range)
		local l = Instance.new("PointLight")
		l.Name = "StealTrail"
		l.Color = color
		l.Range = range or 12
		l.Brightness = 0.7 * glow
		l.Shadows = false
		l.Parent = root
	end

	-- The ribbon, a soft wide glow around it, and a bright core on the better trails
	trail({
		Lifetime = t.Life, WidthScale = NumberSequence.new(t.Width, 0.15),
		Transparency = NumberSequence.new({kp(0, 0.05), kp(0.6, 0.35), kp(1, 1)}),
		LightEmission = (t.Animated and 0.35 or (t.Light or 0) * 0.6) * glow,
	})
	trail({Lifetime = t.Life * 0.8, WidthScale = NumberSequence.new(t.Width * 1.9, 0.3), Transparency = NumberSequence.new(0.86, 1), LightEmission = 0.5 * glow})
	if idx >= 4 then
		trail({
			Lifetime = t.Life * 0.55, WidthScale = NumberSequence.new(t.Width * 0.28, 0), Transparency = NumberSequence.new(0.2, 1), LightEmission = 0.6 * glow,
			Color = t.Animated and colors or ColorSequence.new(Color3.new(1, 1, 1), t.Colors[1]),
		})
	end
	-- Sparkles around you on every trail, more of them the better the trail
	tagRainbow(emitter({
		Texture = TEX_SPARKLE, Color = ColorSequence.new(t.Colors[1], t.Colors[#t.Colors]),
		Size = NumberSequence.new({kp(0, 0), kp(0.3, 0.4 + power * 0.7), kp(1, 0)}), Transparency = NumberSequence.new(0.1, 1),
		Lifetime = NumberRange.new(0.5, 1.1), Rate = 6 + power * 30, Speed = NumberRange.new(0.5, 2),
		Acceleration = Vector3.new(0, 2, 0), Rotation = NumberRange.new(0, 360), RotSpeed = NumberRange.new(-120, 120),
	}))
	if t.Particles then
		tagRainbow(emitter({Texture = TEX_SPARKLE, Color = ColorSequence.new(t.Particles), Size = NumberSequence.new(0.8, 0), Lifetime = NumberRange.new(0.5, 1), Rate = 12 + idx * 2, Speed = NumberRange.new(1, 3)}))
	end

	if t.Fx == "fire" then
		-- real flames licking up off you, with smoke
		emitter({
			Texture = TEX_FIRE, Color = ColorSequence.new(rgb(255, 230, 120), rgb(255, 60, 10)), Size = NumberSequence.new(1.8, 0.2),
			Transparency = NumberSequence.new(0.1, 1), Lifetime = NumberRange.new(0.35, 0.7), Rate = 45, Speed = NumberRange.new(2, 5),
			Acceleration = Vector3.new(0, 9, 0), SpreadAngle = Vector2.new(25, 25), RotSpeed = NumberRange.new(-90, 90), EmissionDirection = Enum.NormalId.Top,
		})
		emitter({Texture = TEX_SMOKE, Color = ColorSequence.new(rgb(60, 30, 20)), Size = NumberSequence.new(1, 3), Transparency = NumberSequence.new(0.65, 1), Lifetime = NumberRange.new(0.8, 1.4), Rate = 8, Speed = NumberRange.new(1, 2), Acceleration = Vector3.new(0, 4, 0), LightEmission = 0})
		light(t.Colors[2] or t.Colors[1])
	elseif t.Fx == "stars" then
		-- big twinkling stars plus little shooting stars streaking off you
		tagRainbow(emitter({Texture = TEX_SPARKLE, Color = ColorSequence.new(Color3.new(1, 1, 1), t.Colors[1]), Size = NumberSequence.new({kp(0, 0), kp(0.5, 1.3), kp(1, 0)}), Lifetime = NumberRange.new(0.8, 1.4), Rate = 14, Speed = NumberRange.new(0.3, 1.5), Rotation = NumberRange.new(0, 360), RotSpeed = NumberRange.new(-60, 60)}))
		emitter({Color = ColorSequence.new(Color3.new(1, 1, 1)), Size = NumberSequence.new(0.25, 0), Orientation = Enum.ParticleOrientation.VelocityParallel, Squash = NumberSequence.new(3), Lifetime = NumberRange.new(0.3, 0.5), Rate = 10, Speed = NumberRange.new(10, 16)})
		light(t.Colors[1], 10)
	elseif t.Fx == "lightning" then
		-- crackling sparks here; the actual bolts are drawn by each player's game
		root:SetAttribute("TrailFx", "lightning")
		root:SetAttribute("TrailColor", (t.Colors[2] or t.Colors[1]):ToHex())
		emitter({Color = ColorSequence.new(Color3.new(1, 1, 1), t.Colors[#t.Colors]), Size = NumberSequence.new(0.18, 0.05), Orientation = Enum.ParticleOrientation.VelocityParallel, Squash = NumberSequence.new(3), Lifetime = NumberRange.new(0.08, 0.18), Rate = 40, Speed = NumberRange.new(12, 22)})
		light(t.Colors[2] or t.Colors[1], 14)
	elseif t.Fx == "snow" then
		emitter({Texture = TEX_SPARKLE, Color = ColorSequence.new(Color3.new(1, 1, 1), rgb(170, 230, 255)), Size = NumberSequence.new(0.6, 0.2), Lifetime = NumberRange.new(1, 2), Rate = 24, Speed = NumberRange.new(0.5, 1.5), Acceleration = Vector3.new(0, -3, 0), RotSpeed = NumberRange.new(-90, 90), LightEmission = 0.6})
		emitter({Texture = TEX_SMOKE, Color = ColorSequence.new(rgb(200, 240, 255)), Size = NumberSequence.new(1.5, 3.5), Transparency = NumberSequence.new(0.75, 1), Lifetime = NumberRange.new(1, 1.6), Rate = 6, Speed = NumberRange.new(0.3, 1), LightEmission = 0.3})
		light(rgb(170, 230, 255), 12)
	elseif t.Fx == "void" then
		-- dark swirling smoke with purple sparks pulled into it
		emitter({Texture = TEX_SMOKE, Color = ColorSequence.new(rgb(120, 40, 220), rgb(10, 0, 20)), Size = NumberSequence.new(1.6, 3.5), Transparency = NumberSequence.new(0.45, 1), Lifetime = NumberRange.new(0.8, 1.3), Rate = 18, Speed = NumberRange.new(0.5, 1), LightEmission = 0, RotSpeed = NumberRange.new(-120, 120)})
		emitter({Texture = TEX_SPARKLE, Color = ColorSequence.new(rgb(220, 140, 255)), Size = NumberSequence.new(0.6, 0), Lifetime = NumberRange.new(0.5, 0.9), Rate = 16, Speed = NumberRange.new(-6, -3), Shape = Enum.ParticleEmitterShape.Sphere})
		light(rgb(150, 60, 255), 12)
	elseif t.Fx == "glow" then
		-- aurora: soft drifting ribbons of light
		emitter({Texture = TEX_SMOKE, Color = colors, Size = NumberSequence.new(2, 4), Transparency = NumberSequence.new(0.7, 1), Lifetime = NumberRange.new(1, 1.6), Rate = 10, Speed = NumberRange.new(0.5, 1.5), Acceleration = Vector3.new(0, 3, 0), LightEmission = 1})
		light(t.Colors[1], 14)
	end
	-- SIGNATURE EFFECT: every trail has one thing of its own
	local down, up = Vector3.new(0, -26, 0), Vector3.new(0, 3, 0)
	local name = t.Name

shopFunction.OnServerInvoke = function(player, action, idx)
	idx = tonumber(idx)
	if action == "unequip" then
		local had = C.TRAILS[player:GetAttribute("TrailEquipped") or 0]
		player:SetAttribute("TrailEquipped", 0)
		applyTrail(player)
		if had and (had.Speed or 1) > 1 then
			notify(player, had.Name .. " Trail removed: speed bonus off", BLUE)
		end
		return true, "Trail removed"
	end
	local t = idx and C.TRAILS[idx]
	if not t then
		return false, "That trail doesn't exist"
	end
	if action == "buy" then
		-- no shopping until the Trail Trader has finished his first-time chat
		local data = playerData[player]
		if not (data and data.TrailIntroDone) then
			return false, "Talk to the Trail Trader first!"
		end
		if ownsTrail(player, idx) then
			return false, "You already own that trail"
		end
		local cash = player.leaderstats.Cash
		if cash.Value < t.Price then
			return false, "You need $" .. fmt(t.Price)
		end
		cash.Value -= t.Price
		local list = ownedTrails(player)
		table.insert(list, idx)
		player:SetAttribute("TrailsOwned", HttpService:JSONEncode(list))
		player:SetAttribute("TrailEquipped", idx)
		applyTrail(player)
		notifyTrailSpeed(player, t)
		return true, "Bought the " .. t.Name .. " trail!"
	elseif action == "equip" then
		if not ownsTrail(player, idx) then
			return false, "You don't own that trail"
		end
		player:SetAttribute("TrailEquipped", idx)
		applyTrail(player)
		notifyTrailSpeed(player, t)
		return true, "Equipped " .. t.Name
	end
	return false, "?"
end

claimIndexFunction.OnServerInvoke = function(player, name)
	local data = playerData[player]
	if not data then return false, "Try again in a moment" end
	local cash, speed, n = 0, 0, 0
	for found, r in pairs(data.Found) do
		local level = C.GAME_LEVEL[found]
		if level and (name == "all" or name == found) and not data.Claimed[found] then
			local reward = C.indexReward(level, r)
			cash += reward.Cash
			speed += reward.Speed
			n += 1
			data.Claimed[found] = true
		end
	end
	if n == 0 then
		return false, "Nothing to claim yet"
	end
	player.leaderstats.Cash.Value += cash
	player:SetAttribute("SpeedPoints", (player:GetAttribute("SpeedPoints") or C.START_SPEED) + speed)
	updateSpeed(player)
	pushIndex(player)
	-- the amounts too, so the client can fly them into the CASH and SPEED boxes
	return true, string.format("Claimed %d game%s: +$%s and +%s speed!", n, n == 1 and "" or "s", fmt(cash), fmt(speed)), cash, speed
end

local function giveGame(player, minLevel, maxLevel, r)
	local level = math.random(minLevel, maxLevel)
	local rec = newGame(level, pickGame(level), r)
	local plot = playerPlots[player]
	local slot = plot and freeSlot(plot)
	if slot then
		placeInSlot(plot, slot, createItem(rec), rec)
	else
		table.insert(heldList(player), {Item = createItem(rec), Data = rec, Kind = "own"})
		markFound(player, rec)
		syncHeld(player)
	end
	return rec
end

					data.Found[name] = 1
					n += 1
				end
			end
			pushIndex(player)

					grabFromVault(player, spot)
					return "grabbed " .. name
				end
			end
			return "nothing on display in level " .. level
		end
		return "unknown command"
	end
end

-- DAILY LOGIN
------------------------------------------------------------------
local function today()
	return math.floor(os.time() / 86400)
end

local function dailyStatus(player)
	local data = playerData[player]
	local last = data.DailyDay or -10
	local t = today()
	if last == t then
		return data.DailyStreak or 1, false
	end
	if last == t - 1 then
		return (data.DailyStreak or 0) % #C.DAILY + 1, true
	end
	return 1, true
end

local function pushDaily(player)
	local idx, ready = dailyStatus(player)
	player:SetAttribute("DailyNext", idx)
	player:SetAttribute("DailyReady", ready)
end

dailyFunction.OnServerInvoke = function(player)
	local data = playerData[player]
	if not data or not data.Loaded then return false, "Try again in a moment" end
	local idx, ready = dailyStatus(player)
	if not ready then
		return false, "Come back tomorrow for Day " .. (idx % #C.DAILY + 1) .. "!"
	end
	local reward = C.DAILY[idx]
	local msg = reward.Text
	if reward.Kind == "cash" then
		player.leaderstats.Cash.Value += reward.Amount
	elseif reward.Kind == "speed" then
		player:SetAttribute("SpeedPoints", (player:GetAttribute("SpeedPoints") or C.START_SPEED) + reward.Amount)
		updateSpeed(player)
	elseif reward.Kind == "game" then
		local rec = giveGame(player, reward.Levels[1], reward.Levels[2], reward.Rarity)
		msg = rec.Rarity .. " " .. rec.Name .. "!"
		if reward.Cash then
			player.leaderstats.Cash.Value += reward.Cash
			msg ..= " + $" .. fmt(reward.Cash)
		end
	end
	data.DailyDay = today()
	data.DailyStreak = idx
	pushDaily(player)
	return true, "Day " .. idx .. " reward: " .. msg
end

local function hasPass(player, key)
	return player:GetAttribute("Pass_" .. key) == true
end

-- Speed and money multipliers = game passes x any active timed boost
local function boostMult(player, stat)
	if (player:GetAttribute("Boost" .. stat .. "Until") or 0) > os.time() then
		return player:GetAttribute("Boost" .. stat) or 1
	end
	return 1
end

local function moneyMult(player)
	local friends = 1 + C.friendBoost(player:GetAttribute("FriendsHere") or 0)
	return (hasPass(player, "DoubleMoney") and 2 or 1) * boostMult(player, "Money") * friends
end

local function refreshMultipliers(player)
	local trail = C.TRAILS[player:GetAttribute("TrailEquipped") or 0]
	local trailSpeed = trail and trail.Speed or 1
	local mult = (hasPass(player, "DoubleSpeed") and 2 or 1) * boostMult(player, "Speed") * trailSpeed
	if player:GetAttribute("SpeedMultiplier") ~= mult then
		player:SetAttribute("SpeedMultiplier", mult)
	end
	-- the speed badge by the HUD shows exactly the trail factor used above
	if player:GetAttribute("TrailSpeed") ~= trailSpeed then
		player:SetAttribute("TrailSpeed", trailSpeed)
	end
	updateSpeed(player)
end
-- a trail's speed applies the moment it's equipped, not on the next base tick
local function watchTrail(player)
	player:GetAttributeChangedSignal("TrailEquipped"):Connect(function()
		refreshMultipliers(player)
	end)
end
Players.PlayerAdded:Connect(watchTrail)
for _, p in ipairs(Players:GetPlayers()) do
	watchTrail(p)
end

local function applyPasses(player)
	refreshMultipliers(player)
end

local function checkPasses(player)
	for _, gp in ipairs(C.GAMEPASSES) do
		if gp.Id ~= 0 then
			local ok, owns = pcall(function()
				return MarketplaceService:UserOwnsGamePassAsync(player.UserId, gp.Id)
			end)
			if ok and owns then
				player:SetAttribute("Pass_" .. gp.Key, true)
			end
		end
	end
	applyPasses(player)
end

end)

local function grantProduct(player, key)
	local data = playerData[player]
	local plot = playerPlots[player]
	local income = plot and plot.LastIncome or 0
	if key == "Starter" then
		player.leaderstats.Cash.Value += 25000
		player:SetAttribute("SpeedPoints", (player:GetAttribute("SpeedPoints") or C.START_SPEED) + 5000)
		updateSpeed(player)
		if not data.StarterBought then
			giveGame(player, 6, 10, 4)
		end
		data.StarterBought = true
		player:SetAttribute("StarterBought", true)
		rewardEvent:FireClient(player, {Title = "STARTER PACK!", Sub = "$25K, +5,000 speed and an Epic game", Color = GOLD:ToHex(), Icon = "\u{1F381}", Kind = "pack", Tier = 4, Badge = "THANK YOU!"})
	elseif key == "Treadmill" then
		local level = player:GetAttribute("TreadLevel") or 0
		if level < C.TREAD_MAX_LEVEL then
			player:SetAttribute("TreadLevel", level + 1)
			if plot then
				rebuildTreadmill(plot)
			end
		end
		notify(player, "Unlocked the " .. TreadModels.get(player:GetAttribute("TreadLevel") or 0).Name .. " treadmill!", GOLD)
	else
		-- a pack's place in its row sets how big its reveal is (the first is tier 2, the fourth and fifth tier 5)
		local function packTier(p)
			local n = 0
			for _, q in ipairs(C.PRODUCTS) do
				if q.Section == p.Section then
					n += 1
					if q == p then
						return math.min(5, n + 1)
					end
				end
			end
			return 2
		end
		for _, p in ipairs(C.PRODUCTS) do
			if p.Key == key and p.Minutes then
				local amount = math.floor(math.max(p.Min or 0, income * 60 * p.Minutes))
				player.leaderstats.Cash.Value += amount
				rewardEvent:FireClient(player, {Title = string.upper(p.Name) .. "!", Sub = "+$" .. fmt(amount), Color = Color3.fromRGB(90, 220, 110):ToHex(), Kind = "cash", Amount = amount, Tier = packTier(p), Badge = "THANK YOU!"})
			elseif p.Key == key and p.Amount then
				-- speed packs give a fixed amount of speed
				local points = player:GetAttribute("SpeedPoints") or C.START_SPEED
				player:SetAttribute("SpeedPoints", points + p.Amount)
				updateSpeed(player)
				rewardEvent:FireClient(player, {Title = string.upper(p.Name) .. "!", Sub = "+" .. fmt(p.Amount) .. " speed", Color = Color3.fromRGB(80, 170, 255):ToHex(), Kind = "speed", Amount = p.Amount, Tier = packTier(p), Badge = "THANK YOU!"})
			elseif p.Key == key and p.Percent then
				local points = player:GetAttribute("SpeedPoints") or C.START_SPEED
				local amount = math.floor(math.max(p.Min or 0, points * p.Percent / 100))
				player:SetAttribute("SpeedPoints", points + amount)
				updateSpeed(player)
				notify(player, "+" .. fmt(amount) .. " speed!", GOLD)
			end
		end
	end
end

local saveNow -- writes the player's save right away and says if it worked (set in SAVING, below)
local trailForProduct = {}
for idx, id in pairs(C.TRAIL_PRODUCTS) do
	if id ~= 0 then
		trailForProduct[id] = idx
	end
end

-- A trail bought with Robux: given exactly like a cash purchase, then saved, and only confirmed once the save worked.
-- Every handled receipt is kept in the save, so a receipt Roblox sends again can never give the trail twice.
local function grantTrailReceipt(player, info, idx)
	local data = playerData[player]
	if not loadedData[player] then
		return Enum.ProductPurchaseDecision.NotProcessedYet -- their save hasn't loaded: Roblox asks again later
	end
	data.Receipts = data.Receipts or {}
	if not table.find(data.Receipts, info.PurchaseId) then
		local t = C.TRAILS[idx]
		if not ownsTrail(player, idx) then
			local list = ownedTrails(player)
			table.insert(list, idx)
			player:SetAttribute("TrailsOwned", HttpService:JSONEncode(list))
		end
		player:SetAttribute("TrailEquipped", idx)
		applyTrail(player)
		table.insert(data.Receipts, info.PurchaseId)
		-- shortcut: only the last 100 receipts are kept, plenty since Roblox retries a receipt within days
		while #data.Receipts > 100 do
			table.remove(data.Receipts, 1)
		end
		rewardEvent:FireClient(player, {Title = "NEW TRAIL!", Sub = t.Name .. "  \u{2022}  \u{00D7}" .. string.format("%g", t.Speed) .. " speed", Color = t.Colors[1]:ToHex(),
			Icon = "\u{2728}", Kind = "trail", Rarity = t.Rarity, Badge = "THANK YOU!"})
		notifyTrailSpeed(player, t)
	end
	-- already given (maybe in an earlier try whose save failed): confirm only once it's safely saved
	if saveNow(player) then
		return Enum.ProductPurchaseDecision.PurchaseGranted
	end
	return Enum.ProductPurchaseDecision.NotProcessedYet
end

-- 9 Robux: the buyer's own base lock goes back to full. The receipt is kept in their save like a trail's, so a
-- receipt Roblox sends again resets nothing twice. No base yet (still joining) = Roblox asks again later.
-- During a Vault theft the lock is reset but stays paused until they're home with the game.
local function grantLockReset(player, info)
	local plot = playerPlots[player]
	local data = playerData[player]
	if not plot or not loadedData[player] then
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end
	data.Receipts = data.Receipts or {}
	if not table.find(data.Receipts, info.PurchaseId) then
		table.insert(data.Receipts, info.PurchaseId)
		while #data.Receipts > 100 do
			table.remove(data.Receipts, 1)
		end
		startLock(plot)
		player:SetAttribute("LockResetAt", plot.LockStamp)
		notify(player, "Base lock reset to " .. Kit.fmtTime(C.LOCK_TIME) .. "!" .. (plot.LockPaused == "theft" and " It stays paused until you're home with your game." or ""), GREEN)
		soundEvent:FireClient(player, "lock")
	end
	return Enum.ProductPurchaseDecision.PurchaseGranted
end

MarketplaceService.ProcessReceipt = function(info)
	local player = Players:GetPlayerByUserId(info.PlayerId)
	if not player or not playerData[player] then
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end
	if trailForProduct[info.ProductId] then
		return grantTrailReceipt(player, info, trailForProduct[info.ProductId])
	end
	for _, p in ipairs(C.PRODUCTS) do
		if p.Id ~= 0 and p.Id == info.ProductId then
			grantProduct(player, p.Key)
			return Enum.ProductPurchaseDecision.PurchaseGranted
		end
	end
	local data = playerData[player]
	if C.SPIN_PRODUCT.Id ~= 0 and info.ProductId == C.SPIN_PRODUCT.Id then
		data.ExtraSpins = (data.ExtraSpins or 0) + 1
		player:SetAttribute("ExtraSpins", data.ExtraSpins)
		notify(player, "Extra spin added! Go spin the Lucky Wheel!", GOLD)
		return Enum.ProductPurchaseDecision.PurchaseGranted
	end
	if C.LOCK_RESET_PRODUCT.Id ~= 0 and info.ProductId == C.LOCK_RESET_PRODUCT.Id then
		return grantLockReset(player, info)
	end
	if C.PLAYTIME_SKIP_PRODUCT.Id ~= 0 and info.ProductId == C.PLAYTIME_SKIP_PRODUCT.Id then
		data.PlayStart = os.time() - 3601
		player:SetAttribute("PlayStart", data.PlayStart)
		notify(player, "All playtime rewards unlocked! Claim them from the gift button.", GOLD)
		return Enum.ProductPurchaseDecision.PurchaseGranted
	end
	for _, tier in ipairs(C.SKIP_PRODUCTS) do
		if tier.Id ~= 0 and info.ProductId == tier.Id then
			local pending = data.PendingSkip
			if pending and not pending.Installed then
				-- Finishes on the next tick so it gets the normal celebration
				pending.Progress = math.max(pending.Progress, pending.InstallTime - 0.5)
			end
			data.PendingSkip = nil
			return Enum.ProductPurchaseDecision.PurchaseGranted
		end
	end
	return Enum.ProductPurchaseDecision.NotProcessedYet
end

-- Friends in the same server boost each other's cash (C.friendBoost: +10% for the first, +2.5% for each one after).
-- It counts friends here right now. Only one refresh runs at a time (IsFriendsWith yields) and it runs again if
-- players came or went meanwhile, so friends joining together never double a boost or a notification.
local friendJoinedEvent = Kit.event("FriendJoined")
local friendsOf = {}  -- player -> {[userId] = display name} of the friends here at the last refresh
local refreshingFriends, refreshFriendsAgain = false, false
local leftServer = setmetatable({}, {__mode = "k"}) -- players on their way out still show in GetPlayers for a moment
local function areFriends(p, q)
	local key = math.min(p.UserId, q.UserId) .. "-" .. math.max(p.UserId, q.UserId)
	if friendCache[key] == nil then
		local ok, yes = pcall(p.IsFriendsWith, p, q.UserId)
		if not ok then
			return false -- not cached, so the next refresh asks again
		end
		friendCache[key] = yes
	end
	return friendCache[key]
end
local function pct(k)
	return (string.format("%.1f", k * 100):gsub("%.0$", "")) .. "%"
end
local function refreshFriends()
	if refreshingFriends then
		refreshFriendsAgain = true
		return
	end
	refreshingFriends = true
	repeat
		refreshFriendsAgain = false
		local list = Players:GetPlayers()
		for _, p in ipairs(list) do
			local here = {}
			for _, q in ipairs(list) do
				if q ~= p and q.Parent and not leftServer[q] and areFriends(p, q) then
					here[q.UserId] = q
				end
			end
			if p.Parent and not leftServer[p] then
				local before = friendsOf[p]
				local now, n = {}, 0
				for id, q in pairs(here) do
					if before and before[id] then
						now[id] = before[id]
						n += 1
					end
				end
				for id, q in pairs(here) do
					if not now[id] then
						now[id] = q.DisplayName
						n += 1
						-- each friend's own share: the first is worth +10%, the rest +2.5%
						friendJoinedEvent:FireClient(p, {UserId = id, Name = q.DisplayName, Gain = C.friendBoost(n) - C.friendBoost(n - 1), Total = C.friendBoost(n), Here = before == nil})
					end
				end
				for id, name in pairs(before or {}) do
					if not now[id] then
						notify(p, name .. " left. Friend Boost is now +" .. pct(C.friendBoost(n)), BLUE)
					end
				end
				friendsOf[p] = now
				p:SetAttribute("FriendsHere", n)
				p:SetAttribute("FriendBoost", C.friendBoost(n))
			end
		end
	until not refreshFriendsAgain
	refreshingFriends = false
end
Players.PlayerAdded:Connect(function()
	-- a moment for their screen to load so the notification is seen
	task.delay(2, refreshFriends)
end)
Players.PlayerRemoving:Connect(function(leaving)
	leftServer[leaving] = true
	friendsOf[leaving] = nil
	-- forget cached friendships with the player who left
	local id = tostring(leaving.UserId)
	for key in pairs(friendCache) do
		local a, b = string.match(key, "^(%d+)%-(%d+)$")
		if a == id or b == id then
			friendCache[key] = nil
		end
	end
	task.defer(refreshFriends)
end)

-- INVITE PAD: every player sees one of their own friends standing on the pad.
-- Roblox doesn't share "best friend" rankings, so this picks a friend who's in
-- this server first, then one who's online, then the first on their friends list.
local inviteAvatars = ReplicatedStorage:FindFirstChild("InviteAvatars") or Instance.new("Folder")
inviteAvatars.Name = "InviteAvatars"
inviteAvatars:ClearAllChildren()
inviteAvatars.Parent = ReplicatedStorage
local function buildInviteAvatar(player)
	local ok, pages = pcall(Players.GetFriendsAsync, Players, player.UserId)
	if not ok or not pages then return end
	local best, bestScore
	for _, f in ipairs(pages:GetCurrentPage()) do
		local score = (Players:GetPlayerByUserId(f.Id) and 2 or 0) + (f.IsOnline and 1 or 0)
		if not best or score > bestScore then
			best, bestScore = f, score
		end
	end
	if not best or not player.Parent then return end
	local made, model = pcall(Players.CreateHumanoidModelFromUserId, Players, best.Id)
	if not made or not model or not player.Parent then return end
	-- Scale first, then anchor ONLY the root. Anchoring every part froze the limbs before the
	-- avatar's body proportions were applied, which stretched necks on some avatars.
	model:ScaleTo(C.INVITE_SCALE)
	for _, d in ipairs(model:GetDescendants()) do
		if d:IsA("BasePart") then
			d.Anchored = d.Name == "HumanoidRootPart"
			d.CanCollide = false
			d.CanQuery = false
		elseif d:IsA("LuaSourceContainer") then
			d:Destroy()
		end
	end
	local hum = model:FindFirstChildOfClass("Humanoid")
	if hum then
		hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
		for _, st in ipairs({Enum.HumanoidStateType.FallingDown, Enum.HumanoidStateType.Ragdoll, Enum.HumanoidStateType.Dead}) do
			hum:SetStateEnabled(st, false)
		end
	end
	model.Name = tostring(player.UserId)
	model:SetAttribute("FriendName", best.DisplayName)
	model.Parent = inviteAvatars
end
Players.PlayerAdded:Connect(function(player)

-- Gives a reward and returns a short description of it (plus the game record for a game, the amount for cash/speed)
local function grantReward(player, reward)
	local plot = playerPlots[player]
	local income = plot and plot.LastIncome or 0
	if reward.Kind == "cash" then
		local amount = math.floor(math.max(reward.Min or 0, income * 60 * (reward.Minutes or 1)))
		player.leaderstats.Cash.Value += amount
		return "+$" .. fmt(amount), nil, amount
	elseif reward.Kind == "speed" then
		local points = player:GetAttribute("SpeedPoints") or C.START_SPEED
		local amount = math.floor(math.max(reward.Min or 0, points * (reward.Percent or 1) / 100))
		player:SetAttribute("SpeedPoints", points + amount)
		updateSpeed(player)
		return "+" .. fmt(amount) .. " speed", nil, amount
	elseif reward.Kind == "boost" then
		local key = "Boost" .. reward.Stat
		local active = (player:GetAttribute(key .. "Until") or 0) > os.time()
		local untilT = math.max(player:GetAttribute(key .. "Until") or 0, os.time()) + reward.Minutes * 60
		-- a new boost never lowers one that's running: the higher multiplier carries on for the combined time
		player:SetAttribute(key, active and math.max(player:GetAttribute(key) or 1, reward.Mult) or reward.Mult)
		player:SetAttribute(key .. "Until", untilT)
		refreshMultipliers(player)
		return string.format("%s\u{00D7} %s for %d minutes", tostring(reward.Mult), reward.Stat, reward.Minutes)
	elseif reward.Kind == "game" then
		local top = bestLevel(player)
		local r = reward.Rarity
		local lo, hi = math.max(1, top - 2), math.min(C.NUM_LEVELS, top + 1)
		if reward.Mystery then
			r = math.random() < 0.3 and 5 or 4
			-- sized to your money: a game earning about (cash / 5000) per second, so it's a real
			-- boost for anyone - never a jackpot for a new player or a dud for a rich one.
			-- Capped at the levels you can earn fully from.
			local target = math.max(1, player.leaderstats.Cash.Value / 5000)
			local cap = math.min(C.NUM_LEVELS, top + C.STEAL_LEVEL_GRACE)
			local L = 1
			while L < cap and C.gameIncome(L + 1, r) <= target do
				L += 1
			end
			lo, hi = math.max(1, L - 1), L
		end
		local rec = giveGame(player, lo, hi, r)
		return rec.Rarity .. " " .. rec.Name, rec
	end
	return ""
end

end

spinFunction.OnServerInvoke = function(player)
	local data = playerData[player]
	if not data or not data.Loaded then return false, "Try again in a moment" end
	if os.time() >= (data.LastSpin or 0) + C.SPIN_COOLDOWN then
		data.LastSpin = os.time()
	elseif (data.ExtraSpins or 0) > 0 then
		data.ExtraSpins -= 1
	else
		-- said on screen too (the board's own message is only readable up close), with when the next free one comes
		notify(player, "No spins available!  Next free spin in " .. Kit.fmtTime((data.LastSpin or 0) + C.SPIN_COOLDOWN - os.time()), RED)
		return false, "No spins available!"
	end
	local total = 0
	for _, r in ipairs(C.SPIN_REWARDS) do
		total += r.Weight
	end
	local roll = math.random() * total
	local idx = #C.SPIN_REWARDS
	for i, r in ipairs(C.SPIN_REWARDS) do
		roll -= r.Weight
		if roll <= 0 then
			idx = i
			break
		end
	end
	local text, rec = grantReward(player, C.SPIN_REWARDS[idx])
	pushSpin(player)
	-- a won game travels with the result so the reveal can show its cover
	return true, idx, text, rec and {Name = rec.Name, Rarity = rec.Rarity, Color = rec.Color:ToHex()}
end

playtimeFunction.OnServerInvoke = function(player, which)
	local data = playerData[player]
	if not data or not data.Loaded then return false, "Try again in a moment" end
	local played = os.time() - data.PlayStart
	local got = {}
	for i, reward in ipairs(C.PLAYTIME) do
		if (which == "all" or which == i) and not data.PlayClaimed[i] and played >= reward.Minute * 60 then
			data.PlayClaimed[i] = true
			local text, rec, amount = grantReward(player, reward)
			table.insert(got, text)
			-- later gifts in the hour are worth more, so they get a bigger reveal (a game's comes from its rarity)
			popup(player, reward.Mystery and "MYSTERY GAME!" or (reward.Minute .. " MINUTE GIFT!"), text, reward.Color, reward.Icon, reward.Mystery and rec,
				{Kind = reward.Kind, Amount = amount, Tier = not rec and math.clamp(math.ceil(i / #C.PLAYTIME * 3), 1, 3) or nil, Badge = "PLAYTIME GIFT"})
		end
	end
	pushPlaytime(player)
	if #got == 0 then
		return false, "Nothing ready yet - keep playing!"
	end
	return true, "Claimed " .. #got .. " gift" .. (#got == 1 and "" or "s") .. "!"
end

local function findOwnedRecord(player, id)
	local plot = playerPlots[player]
	if plot then
		for _, slot in pairs(plot.Slots) do
			if slot.Data and slot.Data.Id == id then
				return slot.Data
			end
		end
	end
	for _, e in ipairs(heldList(player)) do
		if e.Kind == "own" and e.Data.Id == id then
			return e.Data
		end
	end
	return nil
end

local function skipTier(rec)
	local steps = math.max(1, math.ceil((rec.InstallTime - rec.Progress) / C.SKIP_SECONDS_PER_STEP))
	for _, tier in ipairs(C.SKIP_PRODUCTS) do
		if tier.Steps >= steps then
			return tier
		end
	end
	return C.SKIP_PRODUCTS[#C.SKIP_PRODUCTS]
end

skipFunction.OnServerInvoke = function(player, id)
	local rec = findOwnedRecord(player, id)
	if not rec or rec.Installed then
		return false, "That game is already installed"
	end
	-- Admins (and Studio testing) install for free
	if RunService:IsStudio() or table.find(C.ADMIN_USER_IDS or {}, player.UserId) then
		rec.Progress = math.max(rec.Progress, rec.InstallTime - 0.5)
		return true, "free"
	end
	local tier = skipTier(rec)
	if tier.Id == 0 then
		return false, "Instant install isn't set up yet"
	end
	playerData[player].PendingSkip = rec
	return true, tier.Id
end

local function recToSave(rec, slotIndex)
	return {L = rec.L, G = rec.G, R = rec.R, I = rec.Income, T = rec.InstallTime, P = math.floor(rec.Progress), S = slotIndex}

local function collectSave(player)
	local plot = playerPlots[player]
	local data = playerData[player] or {}
	local games = {}
	if plot then
		for i = 1, plot.Cases do
			local slot = plot.Slots[i]
			if slot and slot.Data then
				table.insert(games, recToSave(slot.Data, i))
			end
		end
	end
	for _, e in ipairs(heldList(player)) do
		if e.Kind == "own" then
			table.insert(games, recToSave(e.Data, nil))
		end
	end
	return {
		Cash = player.leaderstats.Cash.Value,
		SpeedPoints = player:GetAttribute("SpeedPoints") or C.START_SPEED,
		TreadLevel = player:GetAttribute("TreadLevel") or 0,
		Upgrades = data.Upgrades or 0,
		Bank = plot and plot.Bank or 0,
		Games = games,
		Trails = ownedTrails(player),
		TrailEquipped = player:GetAttribute("TrailEquipped") or 0,
		Found = data.Found or {},
		Claimed = data.Claimed or {},
		DailyDay = data.DailyDay,
		DailyStreak = data.DailyStreak,
		StarterBought = data.StarterBought,
		SeenIntro = data.SeenIntro,
		TrailIntroDone = data.TrailIntroDone,
		Receipts = data.Receipts,
		LastSpin = data.LastSpin,
		ExtraSpins = data.ExtraSpins,
		AreaClaimed = (function()
			local list = {}
			for zi in pairs(data.AreaClaimed or {}) do
				table.insert(list, zi)
			end
			return list
		end)(),
		TutorialDone = data.TutorialDone,
		BestLevel = data.BestLevel,
		Pets = Pets.save(player),
		SavedAt = os.time(),
		V = C.SAVE_VERSION, -- games and BestLevel use the 40-level vault
	}
end

local function writeSave(player, data)
	if not saveStore then return false end
	local key = "Player_" .. player.UserId
	for attempt = 1, 3 do
		local ok, err = pcall(function()
			saveStore:SetAsync(key, data, {player.UserId})
		end)
		if ok then
			return true
		end
		warn("[Saving] Save failed for " .. player.Name .. " (try " .. attempt .. "): " .. tostring(err))
		task.wait(1)
	end
	return false
end

local function claimPlot(player)
	for _, plot in ipairs(plots) do
		if not plot.Owner then
			plot.Owner = player
			plot.Bank = 0
			plot.LockLeft = 0
			plot.SignLabel.Text = player.DisplayName .. "'s Base"
			playerPlots[player] = plot
			refreshLock(plot)
			local floors, cases, tier = C.baseLayout(playerData[player].Upgrades)
			Base.setTier(plot, tier)
			Base.setLayout(plot, floors, cases, caseCallbacks)
			setPlotOwnerPrompts(plot)
			rebuildTreadmill(plot)
			updateUpgradeLabel(plot)
			return plot
		end
	end
	return nil
end

local function releasePlot(player)
	local plot = playerPlots[player]
	if not plot then return end
	stopTreadmill(plot)
	playerPlots[player] = nil
	for _, slot in pairs(plot.Slots) do
		if slot.Item then
			slot.Item:Destroy()
		end
		slot.Item, slot.Data, slot.Reserved = nil, nil, false
	end
	plot.Owner = nil
	plot.Bank = 0
	plot.LockLeft = 0
	Base.setTier(plot, 1)
	Base.setLayout(plot, 1, C.START_CASES, caseCallbacks)
	setPlotOwnerPrompts(plot)
	plot.SignLabel.Text = "Empty Base"
	plot.PadLabel.Text = "$0"
	rebuildTreadmill(plot)
	updateUpgradeLabel(plot)
	Base.updatePodium(plot, {})
end

	for _, g in ipairs(games) do
		local rec = makeRecord(tonumber(g.L), tonumber(g.G), tonumber(g.I) or 1, tonumber(g.T) or 30, tonumber(g.P) or 0, tonumber(g.R) or 1)
		if rec then
			local slot = g.S and plot.Slots[tonumber(g.S)]
			if slot and not slot.Item then
				placeInSlot(plot, slot, createItem(rec), rec)
			else
				table.insert(leftovers, rec)
			end
		end
	end
	for _, rec in ipairs(leftovers) do
		local slot = freeSlot(plot)
		if slot then
			placeInSlot(plot, slot, createItem(rec), rec)
		else
			table.insert(heldList(player), {Item = createItem(rec), Data = rec, Kind = "own"})
		end
	end
end

local function giveSlap(player)
	local backpack = player:FindFirstChildOfClass("Backpack")
	if not backpack or backpack:FindFirstChild("Slap") then return end
	local tool = Instance.new("Tool")
	tool.Name = "Slap"
	tool.CanBeDropped = false
	tool.ToolTip = "Slap thieves to make them drop your game!"
	-- A big cartoon hand: palm (the handle), four fingers and a thumb
	local skin = Kit.rgb(255, 205, 160)
	local handle = Instance.new("Part")
	handle.Name = "Handle"
	handle.Size = Vector3.new(1.6, 0.5, 1.8)
	handle.Color = skin
	handle.Material = Enum.Material.SmoothPlastic
	handle.CanCollide = false
	handle.Massless = true
	handle.Parent = tool
	tool.GripPos = Vector3.new(0, 0, 0.6)
	local function piece(name, size, offset)
		local p = Instance.new("Part")
		p.Name = name
		p.Size = size
		p.Color = skin
		p.Material = Enum.Material.SmoothPlastic
		p.CanCollide = false
		p.Massless = true
		p.CFrame = handle.CFrame * offset
		local weld = Instance.new("WeldConstraint")
		weld.Part0 = handle
		weld.Part1 = p
		weld.Parent = p
		p.Parent = tool
	end
	for i, len in ipairs({1.1, 1.3, 1.25, 1.0}) do
		piece("Finger", Vector3.new(0.34, 0.42, len), CFrame.new(-0.6 + (i - 1) * 0.4, 0, -0.9 - len / 2 + 0.1))
	end
	piece("Thumb", Vector3.new(0.38, 0.42, 0.9), CFrame.new(0.95, 0, -0.2) * CFrame.Angles(0, math.rad(-40), 0))
	piece("Cuff", Vector3.new(1.8, 0.7, 0.5), CFrame.new(0, 0, 1.0))
	tool:FindFirstChild("Cuff").Color = Kit.rgb(255, 80, 80)
	local lastSlap = 0
	tool.Activated:Connect(function()
		if os.clock() - lastSlap < C.SLAP_COOLDOWN then return end
		lastSlap = os.clock()
		-- everyone else sees your arm swing too (the swinger animates their own on click)
		for _, other in ipairs(Players:GetPlayers()) do
			if other ~= player then
				slapSwingEvent:FireClient(other, player)
			end
		end
		-- the hit lands when the swinging arm reaches the front (matches the client's swing)
		task.wait(C.SLAP_HIT_DELAY)
		local root = getRoot(player)
		if not root then return end
		for _, other in ipairs(Players:GetPlayers()) do
			if other ~= player then
				local otherRoot = getRoot(other)
				if otherRoot and not otherRoot.Anchored then
					local offset = otherRoot.Position - root.Position
					if offset.Magnitude <= C.SLAP_RANGE and root.CFrame.LookVector:Dot(offset.Unit) > 0.2 then
						soundEvent:FireClient(player, "slap")
						soundEvent:FireClient(other, "slap")
						knockEvent:FireClient(other, root.CFrame.LookVector * 70 + Vector3.new(0, 35, 0))
						if stolenCount(other) > 0 then
							dropStolen(other)
							notify(other, player.DisplayName .. " slapped you! You dropped it.", RED)
							notify(player, "You made " .. other.DisplayName .. " drop it!", GREEN)
						end
					end
				end
			end
		end
	end)
	tool.Parent = backpack
end

local function onCharacter(player, char)
	-- A new character means you died or fell: stolen games always go back first
	dropStolen(player)
	char:WaitForChild("HumanoidRootPart")
	local hum = char:WaitForChild("Humanoid")
	Kit.setGroup(char, "Players")
	char.DescendantAdded:Connect(function(d)
		if d:IsA("BasePart") then
			d.CollisionGroup = "Players"
		end
	end)
	updateSpeed(player)
	hum.Died:Connect(function()
		dropStolen(player)
		local plot = playerPlots[player]
		if plot and plot.Runner == player then
			stopTreadmill(plot)
		end
		-- Your own games wait safely until you respawn
		for _, e in ipairs(heldList(player)) do
			clearItemExtras(e.Item)
			e.Item.Parent = holdStore
		end
	end)
	task.wait(0.2)
	giveSlap(player)
	local plot = playerPlots[player]
	if plot and char.Parent then
		char:PivotTo(plot.Spawn)
	end
	task.wait(0.1)
	syncHeld(player)
	applyTrail(player)
end

local function onPlayerAdded(player)
	local stats = Instance.new("Folder")
	stats.Name = "leaderstats"
	local cash = Instance.new("NumberValue")
	cash.Name = "Cash"
	cash.Value = C.START_CASH
	cash.Parent = stats
	local speedStat = Instance.new("NumberValue")
	speedStat.Name = "Speed"
	speedStat.Value = C.START_SPEED
	speedStat.Parent = stats
	stats.Parent = player
	player:SetAttribute("SpeedPoints", C.START_SPEED)
	player:SetAttribute("TreadLevel", 0)
	player:SetAttribute("TrailsOwned", "[]")
	player:SetAttribute("TrailEquipped", 0)
	player:SetAttribute("Held", 0)
	playerData[player] = {Upgrades = 0, Found = {}, Claimed = {}, AreaClaimed = {}, PlayStart = os.time(), PlayClaimed = {}, ExtraSpins = 0}

	local saved, loadedOK = loadSave(player)
	if not player.Parent then return end
	local games = {}
	if loadedOK then
		loadedData[player] = true
		if type(saved) == "table" then
			cash.Value = tonumber(saved.Cash) or C.START_CASH
			player:SetAttribute("SpeedPoints", math.max(C.START_SPEED, tonumber(saved.SpeedPoints) or C.START_SPEED))
			player:SetAttribute("TreadLevel", math.clamp(math.floor(tonumber(saved.TreadLevel) or 0), 0, C.TREAD_MAX_LEVEL))
			playerData[player].Upgrades = math.clamp(math.floor(tonumber(saved.Upgrades) or 0), 0, C.MAX_UPGRADES)
			if type(saved.Trails) == "table" then
				player:SetAttribute("TrailsOwned", HttpService:JSONEncode(saved.Trails))
			end
			player:SetAttribute("TrailEquipped", tonumber(saved.TrailEquipped) or 0)
			local data = playerData[player]
			if type(saved.Found) == "table" then
				for name, r in pairs(saved.Found) do
					data.Found[name] = tonumber(r) or 1
				end
			end
			if type(saved.Claimed) == "table" then
				for name, v in pairs(saved.Claimed) do
					data.Claimed[name] = v == true
				end
			end
			data.DailyDay = tonumber(saved.DailyDay)
			data.DailyStreak = tonumber(saved.DailyStreak)
			data.StarterBought = saved.StarterBought == true
			data.SeenIntro = saved.SeenIntro == true
			-- its own flag: SeenIntro was set the moment the old chat started, so it never meant "finished"
			data.TrailIntroDone = saved.TrailIntroDone == true
			data.Receipts = type(saved.Receipts) == "table" and saved.Receipts or {}
			player:SetAttribute("TrailIntroDone", data.TrailIntroDone)
			data.LastSpin = tonumber(saved.LastSpin)
			data.ExtraSpins = tonumber(saved.ExtraSpins) or 0
			data.TutorialDone = saved.TutorialDone == true
			local legacy = saved.V ~= C.SAVE_VERSION
			if legacy and type(saved.Games) == "table" then
				local kept = {}
				for _, g in ipairs(saved.Games) do
					if type(g) == "table" and migrateGame(g) then
						table.insert(kept, g)
					end
				end
				saved.Games = kept
			end
			data.BestLevel = tonumber(saved.BestLevel)
			if data.BestLevel and legacy then
				data.BestLevel = C.LEGACY_BEST[math.clamp(math.floor(data.BestLevel), 1, 50)]
			end
			if data.BestLevel then
				data.BestLevel = math.clamp(math.floor(data.BestLevel), 1, C.NUM_LEVELS)
			end
			if not data.BestLevel then
				-- older saves: trust the deepest game they already own
				data.BestLevel = 1
				if type(saved.Games) == "table" then
					for _, g in ipairs(saved.Games) do
						data.BestLevel = math.max(data.BestLevel, tonumber(g.L) or 1)
					end
				end
			end
			player:SetAttribute("BestLevel", data.BestLevel)
			if type(saved.AreaClaimed) == "table" then
				for _, zi in ipairs(saved.AreaClaimed) do
					data.AreaClaimed[tonumber(zi)] = true
				end
			end
			player:SetAttribute("StarterBought", data.StarterBought)
			games = type(saved.Games) == "table" and saved.Games or {}
			-- Installing keeps going while you're away, at half speed
			local away = math.clamp(os.time() - (tonumber(saved.SavedAt) or os.time()), 0, 60 * 60 * 24 * 30)
			-- ...and every game in a case earns offlineCashRate x its income for the time it was installed (one that
			-- finished while you were away earns from then on). Boosts and friend bonuses don't count offline.
			local finished, offline = 0, 0
			for _, g in ipairs(games) do
				local p, t = tonumber(g.P) or 0, tonumber(g.T) or 30
				if p < t then
					g.P = p + away * C.OFFLINE_INSTALL_RATE
					if g.P >= t then
						finished += 1
					end
				end
				if g.S and tonumber(g.L) then
					local earning = p >= t and away or math.max(0, away - (t - p) / C.OFFLINE_INSTALL_RATE)
					offline += (tonumber(g.I) or 0) * levelFactor(player, {L = tonumber(g.L)}) * earning
				end
			end
			offline = math.floor(offline * offlineCashRate(player))
			if offline >= 1 then
				cash.Value += offline
				data.OfflineEarned = offline -- shown the first time they step into their base
			end
			if finished > 0 then
				task.delay(4, notify, player, finished .. " game" .. (finished == 1 and "" or "s") .. " finished installing while you were away!", GREEN)
			end
			playerData[player].Bank = tonumber(saved.Bank) or 0
		end
	else
		task.delay(3, notify, player, "Couldn't load your save. Progress won't be saved this time.", RED)
	end
	updateSpeed(player)

	local plot = claimPlot(player)
	if plot then
		plot.Bank = playerData[player].Bank or 0
		restoreGames(player, plot, games)
	end
	playerData[player].Loaded = true
	pushIndex(player)
	pushDaily(player)
	pushSpin(player)
	pushPlaytime(player)
	pushAreas(player)
	Pets.load(player, loadedOK and type(saved) == "table" and saved.Pets or nil)
	player:SetAttribute("TutorialDone", playerData[player].TutorialDone == true)
	task.spawn(checkPasses, player)
	if not plot then
		task.delay(3, notify, player, "No free base on this server, sorry!", RED)
	end
	player.CharacterAdded:Connect(function(char)
		onCharacter(player, char)
	end)
	if player.Character then
		task.spawn(onCharacter, player, player.Character)
	end
end

-- the Game Vault: the hall and its rooms, past the entrance
local function inVaultArea(pos)
	return math.abs(pos.X) < C.ROOM_W / 2 + 12 and pos.Z > C.HALL_START_Z - 6
end

				local inVault = inVaultArea(pos)
				if flat.Magnitude > C.FENCE_R - 3 and not inVault and not inPetLab(p) then
					local back = center + flat.Unit * (C.FENCE_R - 14)
					p.Character:PivotTo(CFrame.lookAt(Vector3.new(back.X, 5, back.Z), Vector3.new(center.X, 5, center.Z)))
					r.AssemblyLinearVelocity = Vector3.zero
				end
			end
		end
	end
end)

------------------------------------------------------------------
-- TOP 10 LEADERBOARDS (Money and Speed)
-- Global: saved in ordered data stores, so rankings cover every server and survive
-- restarts. Each player's value is written every minute (and when they leave); the
-- top 10 is read back every minute. Values come from the server only - never the client.
-- Ordered data stores only hold whole numbers, so values are stored as log10 x 1e9
-- (keeps the order exact and fits even huge numbers).
------------------------------------------------------------------
do
	local LB_EVERY = 60
	local stores = {}
	pcall(function()
		stores.Cash = DataStoreService:GetOrderedDataStore("StealAGameTop_Cash_v2")
		stores.Speed = DataStoreService:GetOrderedDataStore("StealAGameTop_Speed_v2")
	end)
	local function encode(v)
		return math.floor(math.log10(math.max(1, v)) * 1e9)
	end
	local function decode(n)
		return 10 ^ (n / 1e9)
	end
	local names = {}
	local function nameOf(userId)
		if names[userId] then
			return names[userId]
		end
		local p = Players:GetPlayerByUserId(userId)
		if p then
			names[userId] = p.DisplayName
			return p.DisplayName
		end
		local ok, n = pcall(Players.GetNameFromUserIdAsync, Players, userId)
		names[userId] = ok and n or "Player"
		return names[userId]
	end
	local function valuesOf(p)
		local ls = p:FindFirstChild("leaderstats")
		if not ls then return nil end
		return {Cash = ls:FindFirstChild("Cash") and ls.Cash.Value or 0, Speed = ls:FindFirstChild("Speed") and ls.Speed.Value or 0}
	end
	local function write(p)
		local data = playerData[p]
		local v = valuesOf(p)
		if not (data and data.Loaded and v) then return end
		for key, store in pairs(stores) do
			pcall(function()
				store:SetAsync("U" .. p.UserId, encode(v[key]))
			end)
		end
	end
	local board = ReplicatedStorage:FindFirstChild("LeaderboardData") or Instance.new("StringValue")
	board.Name = "LeaderboardData"
	board.Parent = ReplicatedStorage
	local function refresh()
		local result = {Global = true}
		for _, key in ipairs({"Cash", "Speed"}) do
			local list = {}
			local store = stores[key]
			local ok, pages = false, nil
			if store then
				ok, pages = pcall(function()
					return store:GetSortedAsync(false, 10)
				end)
			end
			if ok and pages then
				for rank, entry in ipairs(pages:GetCurrentPage()) do
					local uid = tonumber(string.sub(entry.key, 2)) or 0
					list[rank] = {Rank = rank, Name = nameOf(uid), Value = decode(entry.value)}
				end
			else
				-- saving is unavailable: honestly show this server only
				result.Global = false
				local all = {}
				for _, p in ipairs(Players:GetPlayers()) do
					local v = valuesOf(p)
					if v then
						table.insert(all, {Name = p.DisplayName, Value = v[key]})
					end
				end
				table.sort(all, function(a, b) return a.Value > b.Value end)
				for rank = 1, math.min(10, #all) do
					all[rank].Rank = rank
					list[rank] = all[rank]
				end
			end
			result[key] = list
		end
		board.Value = HttpService:JSONEncode(result)
	end
	Players.PlayerRemoving:Connect(write)
	task.spawn(function()
		task.wait(8)
		while true do
			for _, p in ipairs(Players:GetPlayers()) do
				write(p)
			end
			refresh()
			task.wait(LB_EVERY)
		end
	end)
end

local function libraryEntry(rec, where)
	local tier = not rec.Installed and skipTier(rec)
	return {
		Id = rec.Id, SkipRobux = tier and tier.Steps * C.SKIP_ROBUX_PER_STEP or nil,
		Name = rec.Name, Rarity = rec.Rarity, Color = rec.Color:ToHex(), Level = rec.L, Where = where,
		Income = rec.Installed and math.floor(rec.Income * (rec.Factor or 1)) or nil,
		Progress = math.floor(rec.Progress), InstallTime = rec.InstallTime,
	}
end

-- Every second: installs, income, podiums, the Downloads/Installed lists, restocks
local lastDay = today()
task.spawn(function()
	while true do
		task.wait(1)
		local now = os.clock()
		-- a new day: everyone's daily reward unlocks without rejoining
		if today() ~= lastDay then
			lastDay = today()
			for p, d in pairs(playerData) do
				if d.Loaded and p.Parent then
					pushDaily(p)
				end
			end
		end
		for _, plot in ipairs(plots) do
			local owner = plot.Owner
			if owner then
				local income = 0
				local installing, installed, ranked = {}, {}, {}
				local function tick(rec, item, where)
					local factor = levelFactor(owner, rec)
					if factor ~= (rec.Factor or 1) then
						rec.Factor = factor
						if item and rec.Installed then
							setItemLabel(item, rec, where == "In your hands" and "held" or "case")
						end
					end
					-- games only install inside a case, never while you're carrying them
					if not rec.Installed and where ~= "In your hands" then
						rec.Progress += C.ONLINE_INSTALL_RATE * Pets.installMult(owner)
						if rec.Progress >= rec.InstallTime then
							rec.Progress = rec.InstallTime
							rec.Installed = true
							installedEvent:FireClient(owner, {Name = rec.Name, Rarity = rec.Rarity, R = rec.R, Color = rec.Color:ToHex(), Income = math.floor(rec.Income * (rec.Factor or 1))})
							if item then
								celebrate(item, rec, where ~= "In your hands")
							end
						end
						if item then
							setItemLabel(item, rec, where == "In your hands" and "held" or "case")
						end
					end
					if rec.Installed then
						table.insert(installed, libraryEntry(rec, where))
					else
						table.insert(installing, libraryEntry(rec, where))
					end
				end
				for i = 1, plot.Cases do
					local slot = plot.Slots[i]
					if slot and slot.Data then
						local rec = slot.Data
						local was = rec.Installed
						tick(rec, slot.Item, "Case " .. i)
						if rec.Installed then
							income += rec.Income * (rec.Factor or 1)
							table.insert(ranked, rec)
							if not was then
								refreshSlotPrompts(plot, slot)
							end
						end
					end
				end
				for _, e in ipairs(heldList(owner)) do
					if e.Kind == "own" then
						tick(e.Data, e.Item, "In your hands")
					end
				end
				income *= moneyMult(owner)
				refreshMultipliers(owner)
				plot.LastIncome = income
				owner:SetAttribute("IncomePerSec", income)
				plot.Bank += income
				if os.clock() < (plot.PadShowZeroUntil or 0) then
					plot.PadLabel.Text = "$0\n+$" .. fmt(income) .. "/s"
				else
					plot.PadLabel.Text = "$" .. fmt(plot.Bank) .. "\n+$" .. fmt(income) .. "/s"
					if plot.PadColor then
						plot.PadLabel.TextColor3 = plot.PadColor
					end
				end

				table.sort(ranked, function(a, b) return a.Income * (a.Factor or 1) > b.Income * (b.Factor or 1) end)
				local top = {}
				for r = 1, math.min(3, #ranked) do
					local rec = ranked[r]
					top[r] = {Name = rec.Name, Rarity = rec.Rarity, Color = rec.Color, IncomeText = "+$" .. fmt(rec.Income * (rec.Factor or 1)) .. "/s"}
				end
				Base.updatePodium(plot, top)

				table.sort(installing, function(a, b) return (a.InstallTime - a.Progress) < (b.InstallTime - b.Progress) end)
				table.sort(installed, function(a, b) return a.Income > b.Income end)
				libraryEvent:FireClient(owner, {Installing = installing, Installed = installed, IncomePerSecond = income})
			end

			-- Lock timer (see refreshLock): settled every second, which also picks up chases starting and ending
			refreshLock(plot)
			if plot.LockLeft > 0 then
				plot.LockLabel.Text = (plot.LockPaused and '<font color="#ffd23c">LOCKED (PAUSED)</font>\n' or "LOCKED\n") .. Kit.fmtTime(math.ceil(plot.LockLeft))
			end
			-- Lock walls: solid for everyone except the owner (the owner's game makes them passable)
			local locked = plot.LockLeft > 0
			if not locked then
				plot.LockLabel.Text = "LOCK BASE"
			end
			if plot.UnlockPrompt then
				-- only reachable while the base is locked (the owner-only tag decides who sees it)
				plot.UnlockPrompt.MaxActivationDistance = locked and 8 or 0
				plot.UnlockPrompt:SetAttribute("OwnerId", plot.Owner and plot.Owner.UserId or 0)
			end
			for _, b in ipairs(Base.barriers(plot)) do
				b.Transparency = locked and 0.45 or 1
				b.CanCollide = locked
				b:SetAttribute("OwnerId", plot.Owner and plot.Owner.UserId or 0)
			end
			-- so a would-be thief's Steal row can say the base is locked (onCase refuses either way)
			for _, slot in pairs(plot.Slots) do
				slot.MainPrompt:SetAttribute("Locked", locked)
			end
		end

		-- Every 7 minutes, every level swaps to 3 brand-new random games
		if now >= nextRotation then
			nextRotation = now + C.ROTATE_TIME
			for _, spot in ipairs(spots) do
				if not spot.Carried then
					if spot.Item then
						spot.Item:Destroy()
						spot.Item, spot.Data = nil, nil
					end
					spawnSpot(spot)
				end
			end
		end
		local rotateText = '<font color="#5aff78">NEW GAMES IN ' .. Kit.fmtTime(nextRotation - now) .. '</font>'
		for _, label in pairs(world.RestockLabels) do
			label.Text = rotateText
		end
	end
end)

-- Every frame: treadmill belts. Every 0.1s: training, deliveries, locks.
local checkTimer = 0
RunService.Heartbeat:Connect(function(dt)
	for _, plot in ipairs(plots) do
		local runner = plot.Runner
		if runner and #plot.Slats > 0 then
			-- faster every treadmill level (6 to 16 studs/s); kept under the speed where the slats seem to strobe
			local beltSpeed = 6 + 0.25 * (runner:GetAttribute("TreadLevel") or 0)
			local span = Base.SLAT_COUNT * Base.SLAT_GAP
			plot.SlatOffset = (plot.SlatOffset + beltSpeed * dt) % span
			for i, slat in ipairs(plot.Slats) do
				local z = Base.BELT_START + ((i - 1) * Base.SLAT_GAP + plot.SlatOffset) % span
				slat.CFrame = plot.BeltCF * CFrame.new(0, 0, z)
			end
		end
	end

	checkTimer += dt
	if checkTimer < 0.1 then return end
	local step = checkTimer
	checkTimer = 0

	for _, plot in ipairs(plots) do
		local runner = plot.Runner
		if runner then
			if not runner.Parent or not getRoot(runner) or runner ~= plot.Owner then
				stopTreadmill(plot)
			else
				-- exactly one fixed gain per full second on the belt, so the rate never wobbles
				plot.TreadTime = (plot.TreadTime or 0) + step
				while plot.TreadTime >= 1 do
					plot.TreadTime -= 1
					local gain = C.treadGain(runner:GetAttribute("TreadLevel") or 0)
					local points = runner:GetAttribute("SpeedPoints") or C.START_SPEED
					runner:SetAttribute("SpeedPoints", math.min(1e30, points + gain))
					updateSpeed(runner)
					treadGainEvent:FireClient(runner, gain)
				end
			end
		end
	end

	-- Bringing a stolen game into your own base is what steals it: it's yours from that moment (the owner's alarm stops),
	-- and it drops into a free case if you have one, or stays in your hands until you make room
	local clock = os.clock()
	for player, list in pairs(held) do
		if #list > 0 and stolenCount(player) > 0 then
			local root = getRoot(player)
			local myPlot = playerPlots[player]
			-- Fell off the island with a stolen game: it goes back, no free steals
			if root and root.Position.Y < C.FALL_RETURN_Y then
				dropStolen(player)
				notify(player, "You fell off! The game was returned.", RED)
				if myPlot then
					player.Character:PivotTo(myPlot.Spawn)
				end
			elseif root and myPlot and Base.insidePlot(myPlot, root.Position) then
				local changed = false
				for i = #list, 1, -1 do
					local e = list[i]
					if e.Kind ~= "own" then
						local slot = freeSlot(myPlot)
						if slot then
							table.remove(list, i)
							finalizeStolen(player, e)
							placeInSlot(myPlot, slot, e.Item, e.Data)
							changed = true
						else
							finalizeStolen(player, e) -- now an ordinary game in your hands: place it in a case when one is free
							changed = true
							if clock - (fullWarned[player] or 0) > 5 then
								fullWarned[player] = clock
								notify(player, "It's yours! Your cases are full, so you're holding it. Sell a game to make room.", GOLD)
							end
						end
					end
				end
				if changed then
					syncHeld(player)
				end
			end
		end
	end

	-- Offline earnings (already added at join): shown the first time you step into your own base
	for player, data in pairs(playerData) do
		if data.OfflineEarned then
			local root = getRoot(player)
			local myPlot = playerPlots[player]
			if root and myPlot and Base.insidePlot(myPlot, root.Position) then
				offlineEvent:FireClient(player, data.OfflineEarned, offlineCashRate(player))
				data.OfflineEarned = nil
			end
		end
	end

	for _, plot in ipairs(plots) do
		if plot.Owner and plot.LockLeft > 0 then
			for _, player in ipairs(Players:GetPlayers()) do
				if player ~= plot.Owner then
					local root = getRoot(player)
					-- Only for someone already deep inside when the lock went on (the walls stop everyone else)
					if root and Base.insideCore(plot, root.Position) then
						-- anything they took from this base goes back to its case first: it's carried on their body, so the
						-- teleport below would otherwise take it out through the lock walls with them
						local list = heldList(player)
						local gaveBack = false
						for i = #list, 1, -1 do
							local e = list[i]
							if e.Kind == "base" and e.FromPlot == plot then
								table.remove(list, i)
								returnEntry(e)
								gaveBack = true
							end
						end
						if gaveBack then
							syncHeld(player)
						end
						player.Character:PivotTo(plot.CF * CFrame.new(0, 4, -C.PLOT_D / 2 - 10))
						notify(player, "That base is locked!", RED)
					end
				end
			end
		end
	end
end)
