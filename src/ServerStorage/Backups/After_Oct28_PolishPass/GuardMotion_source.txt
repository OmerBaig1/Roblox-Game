--[[
	GuardMotion: draws every shopkeeper's movement on this screen.

	Each guard is a rigged, sculpted body (see Guards.ROSTER on the server) with six joints: Root, Neck,
	LeftShoulder, RightShoulder, LeftHip, RightHip. The server moves the giants (Humanoid:MoveTo) and names the guard
	in the Motion attribute; MOTION[name] here is that guard's own idle, chase gait and catch reaction.

	Moving vs standing: the gait's weight follows the guard's real, smoothed ground speed with hysteresis (on above
	MOVE_ON body lengths a second, off below MOVE_OFF), so a guard standing at its post always plays its idle, never
	its run. The stride phase advances by the distance really covered, so feet never slide or run in place. Idles
	never step: they breathe, shift weight, look around and fidget.
	Joints are written through Transform every PreSimulation. Only guards near the camera are animated.
]]

local RunService = game:GetService("RunService")

local V = Vector3.new
local ZERO = Vector3.zero
local sin, cos, abs, pi, clamp, max = math.sin, math.cos, math.abs, math.pi, math.clamp, math.max
local NEAR = 260 -- studs from the camera within which guards are animated
local MOVE_ON, MOVE_OFF = 1.2, 0.5 -- body lengths a second

local JOINTS = {Root = "Root", Neck = "Neck", LS = "LeftShoulder", RS = "RightShoulder", LH = "LeftHip", RH = "RightHip"}
local KEYS = {"Root", "Neck", "LS", "RS", "LH", "RH"}

-- A pose: joint key -> Vector3 of angles (radians: X pitch, Y yaw, Z roll), plus Off = root offset in body studs.
-- Signs: +X on a hip or shoulder swings the limb forward; -X on Root leans forward; on the left shoulder -Z lifts the
-- arm out to the side (+Z on the right).
local function newPose()
	return {Root = ZERO, Neck = ZERO, LS = ZERO, RS = ZERO, LH = ZERO, RH = ZERO, Off = ZERO}
end
local function legs(p, ph, amp, spread)
	local a = sin(ph)
	p.LH = V(amp * a, 0, -(spread or 0))
	p.RH = V(-amp * a, 0, spread or 0)
end
local function swing(p, ph, amp, out, fwd)
	local a = sin(ph)
	p.LS = V((fwd or 0) - amp * a, 0, -(out or 0.1))
	p.RS = V((fwd or 0) + amp * a, 0, out or 0.1)
end
local function arms(p, l, r)
	p.LS, p.RS = l, r
end
local function breathe(p, t, amp)
	p.Off = V(0, 0.03 * sin(t * 1.6) * (amp or 1), 0)
	p.LS = V(0, 0, -0.08)
	p.RS = V(0, 0, 0.08)
end
local function shake(age, speed, amp)
	return sin(age * (speed or 18)) * (amp or 0.08)
end

	end,
	Catch = function(p, age)
		arms(p, V(0.5, 0, -0.3), V(0.5, 0, 0.3))

		p.LH = V(1.2 * max(0, a) - 0.25, 0, 0)
		p.RH = V(1.2 * max(0, -a) - 0.25, 0, 0)
		swing(p, ph, 0.5, 0.25)
		p.Off = V(0, 0.55 * abs(a), 0)
	end,
	Idle = function(p, t)
		breathe(p, t, 1.2)
		local twitch = (t % 3.1 < 0.12) and 0.15 or 0
		p.RS = V(0.2 + twitch, 0, 0.25)
		p.LS = V(0.2, 0, -0.25)
	end,
	Catch = function(p, age)
		local a = abs(sin(age * 8))
		p.RS = V(2.0 + 0.8 * a, 0, 0.2)
		p.LS = V(0, 0, -0.3)
		p.Off = V(0, 0.8 * a, 0)
	end}
-- Moe: a crouched scurry, claws digging at the air. Idle: squints and sniffs, claws rubbing together. Catch: digs in a frenzy of delight.

-- Queen Lumina: grand jetes, arms in a perfect arc, a pirouette now and then. Idle: a slow, regal pirouette on the spot.

-- Chronos: moves in skipped frames: freezes, jumps ahead, rewinds a little. Idle: turns his hourglass over, impatient.

MOTION.chronos = {Stride = 4.4,
	Gait = function(p, ph, s)
		local glitch = (s.T * 3) % 1
		local q = math.floor((ph + (glitch < 0.15 and -0.6 or 0)) / 0.7) * 0.7
		legs(p, q, 0.65, 0)
		swing(p, q, 0.4, 0.1)

		p.Off = V((glitch < 0.15) and 0.4 or 0, 0.2 * abs(cos(q)), 0)
	end,
	Idle = function(p, t)
		local f = (t % 4) / 4
		p.RS = V(1.2, f < 0.25 and f / 0.25 * pi or pi, 0.2)

		p.Neck = V(-0.3, 0, 0)

		p.Root = V(0, -math.floor(age * 10) * 0.6, 0)

-- the taunt when a chase starts: points straight at you
local function taunt(p, age)
	p.RS = V(1.57, 0, 0.1)
	p.Neck = V(0, 0, 0.1 * sin(age * 10))
end

------------------------------------------------------------------
-- RUNNING THEM
------------------------------------------------------------------
local guards = {} -- model -> state
local function track(model)
	if guards[model] or not model:IsA("Model") then return end
	guards[model] = {Model = model, Phase = 0, MoveW = 0, ActW = 0, Speed = 0, Moving = false, Seed = math.random() * 100}
end
local function untrack(model)
	local st = guards[model]
	if st and st.Joints then
		for _, j in pairs(st.Joints) do
			if j.Parent then
				j.Transform = CFrame.identity
			end
		end
	end
	guards[model] = nil
end
local function findJoints(st)
	local joints = {}
	for _, d in ipairs(st.Model:GetDescendants()) do
		if d:IsA("Motor6D") or d:IsA("AnimationConstraint") then
			for key, name in pairs(JOINTS) do
				if d.Name == name then
					joints[key] = d
				end
			end
		end
	end
	if joints.Root then
		st.Joints = joints
	end
end

local function lerpPose(a, b, w)
	for _, k in ipairs(KEYS) do
		a[k] = a[k]:Lerp(b[k], w)
	end
	a.Off = a.Off:Lerp(b.Off, w)
end

local function step(dt)
	local cam = workspace.CurrentCamera
	local camPos = cam and cam.CFrame.Position or Vector3.zero
	local now = workspace:GetServerTimeNow()
	local t = os.clock()
	for model, st in pairs(guards) do
		local root = model.Parent and model.PrimaryPart
		if not root then
			continue
		end
		if (root.Position - camPos).Magnitude > NEAR then
			st.Far = true
			continue
		end
		if not st.Joints then
			findJoints(st)
			if not st.Joints then
				continue
			end
		end
		local scale = model:GetAttribute("Scale") or 4
		local m = MOTION[model:GetAttribute("Motion")] or MOTION.brutus

		-- how far they really moved since last frame (a teleport home or a stream-in doesn't count)
		local pos = root.Position
		local moved = st.Last and Vector3.new(pos.X - st.Last.X, 0, pos.Z - st.Last.Z).Magnitude or 0
		st.Last = pos
		if st.Far or moved > 60 * scale * dt + 20 then
			moved = 0
			st.Far = false
		end
		-- smoothed speed in body lengths a second, with hysteresis between standing and moving
		local speed = moved / math.max(dt, 1 / 240) / scale
		st.Speed += (speed - st.Speed) * (1 - math.exp(-dt * 10))
		if st.Moving and st.Speed < MOVE_OFF then
			st.Moving = false
		elseif not st.Moving and st.Speed > MOVE_ON then
			st.Moving = true
		end
		st.MoveW += ((st.Moving and 1 or 0) - st.MoveW) * (1 - math.exp(-dt * 7))
		local stride = m.Stride * scale
		local before = st.Phase
		st.Phase = (st.Phase + moved / stride * 2 * pi) % (4 * pi)
		-- a footstep each time the stride passes a half cycle (the footstep sound lives in StealAGameClient)
		if st.Moving and math.floor(before / pi) ~= math.floor(st.Phase / pi) then
			model:SetAttribute("StepTick", (model:GetAttribute("StepTick") or 0) + 1)
		end

		local s = {T = t + st.Seed}
		local pose = newPose()
		m.Idle(pose, t + st.Seed)
		if st.MoveW > 0.01 then
			local gp = newPose()
			m.Gait(gp, st.Phase, s)
			lerpPose(pose, gp, st.MoveW)
		end
		-- one-off reactions, eased in and out
		local act = model:GetAttribute("Act")
		local age = now - (model:GetAttribute("ActAt") or 0)
		local dur = act == "catch" and 1.8 or act == "taunt" and 0.9 or 0
		st.ActW += (((age >= 0 and age < dur) and 1 or 0) - st.ActW) * (1 - math.exp(-dt * 12))
		if st.ActW > 0.01 then
			local ap = newPose()
			if act == "catch" then
				m.Catch(ap, math.max(0, age))
			else
				taunt(ap, math.max(0, age))
			end
			lerpPose(pose, ap, st.ActW)
		end

		for key, joint in pairs(st.Joints) do
			local r = pose[key]
			if key == "Root" then
				joint.Transform = CFrame.new(pose.Off * scale) * CFrame.fromEulerAnglesXYZ(r.X, r.Y, r.Z)
			else
				joint.Transform = CFrame.fromEulerAnglesXYZ(r.X, r.Y, r.Z)
			end
		end
	end
end

task.spawn(function()
	local map = workspace:WaitForChild("StealAGameMap", 60)
	local folder = map and map:WaitForChild("Shopkeepers", 60)
	if not folder then return end
	for _, m in ipairs(folder:GetChildren()) do
		track(m)
	end
	folder.ChildAdded:Connect(track)
	folder.ChildRemoved:Connect(untrack)
	RunService.PreSimulation:Connect(step)
end)
